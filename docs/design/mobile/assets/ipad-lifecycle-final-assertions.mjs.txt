import assert from 'node:assert/strict';
import {readFile,writeFile,stat} from 'node:fs/promises';
import {createHash} from 'node:crypto';
import {join} from 'node:path';
import {AppwireClient} from '/tmp/evener-notifications-consumer.yrcaOf/node_modules/@evener/appwire-client/dist/index.js';
import WebSocket from '/tmp/evener-notifications-consumer.yrcaOf/node_modules/ws/index.js';
const root=process.argv[2]??'.'; const userMarker=process.env.EVENER_EXPECTED_A_USER_MARKER??'owned ipad a held turn 829c'; const agentMarker=process.env.EVENER_EXPECTED_A_AGENT_MARKER??'fixture-A-held-1';
const files=[]; async function readJSON(p){return JSON.parse(await readFile(p,'utf8'));} async function save(name,value){const p=join(root,'run',`cleanup-${name}.json`);await writeFile(p,JSON.stringify(value,null,2)+'\n',{flag:'wx',mode:0o600});files.push({path:p,sha256:createHash('sha256').update(await readFile(p)).digest('hex'),mode:(await stat(p)).mode&0o777});return p;}
async function readHub(label){const d=await readJSON(join(root,label,'runtime-restarted.json')).catch(()=>readJSON(join(root,label,'runtime.json')));const b=await readJSON(join(root,label,'binding.json'));const baseline=await readJSON(join(root,label,'baseline.json'));const token=(await readFile(d.tokenFile,'utf8')).trim();const hub=new AppwireClient({url:d.rpcURL,clientInfo:{name:'ipad-two-hub-cleanup-assertions',version:'1'},socketFactory:u=>new WebSocket(u,{headers:{Authorization:`Bearer ${token}`}})});try{await hub.connect();const read=await hub.request('thread/read',{ref:b.ref,includeTurns:true,itemLimit:40,itemsView:'full',subscribe:false});assert.equal(read.thread.evener.ref,b.ref);assert.equal(read.thread.evener.instanceId,b.instanceId);assert(read.thread.turns.every(t=>t.status==='completed'),'all turns must be completed');const items=read.thread.turns.flatMap(t=>t.items??[]);const users=items.filter(i=>i.type==='userMessage').map(i=>i.text);const agents=items.filter(i=>i.type==='agentMessage').map(i=>i.text);if(label==='A'){assert(users.includes(userMarker),'A held user marker missing');assert(agents.includes(agentMarker),'A held provider reply missing');assert(users.filter(t=>t.startsWith('owned ipad')).every(t=>t===userMarker),'A draft prefix leaked into server user messages');}else{const baseItems=baseline.thread.turns.flatMap(t=>t.items??[]);assert.deepEqual(users,baseItems.filter(i=>i.type==='userMessage').map(i=>i.text),'B user transcript changed');assert.deepEqual(agents,baseItems.filter(i=>i.type==='agentMessage').map(i=>i.text),'B agent transcript changed');}await save(`${label}-full`,read);return {hubPort:d.hubPort,ref:b.ref,instanceId:b.instanceId,turns:read.thread.turns.length,userMessageCount:users.length,agentMessageCount:agents.length};}finally{hub.close();}}
const A=await readHub('A');const B=await readHub('B');const receipt={status:'passed',assertions:['A exact held user/reply present','B user/agent projections byte-equivalent to baseline','all returned turns completed','unsent draft prefix absent from server user messages except owned held marker'],hubs:{A,B},files};await save('receipt',receipt);console.log(JSON.stringify({status:'passed',hubs:{A:{turns:A.turns,userMessageCount:A.userMessageCount,agentMessageCount:A.agentMessageCount},B:{turns:B.turns,userMessageCount:B.userMessageCount,agentMessageCount:B.agentMessageCount}}}));
