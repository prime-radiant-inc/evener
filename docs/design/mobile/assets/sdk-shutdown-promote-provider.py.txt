import json, pathlib, sys, time
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
root=pathlib.Path(sys.argv[1]); port=int(sys.argv[2]); release=root/'release'; records=root/'provider-requests.json'; state=[]
class H(BaseHTTPRequestHandler):
  def log_message(self,*a): pass
  def do_GET(self):
    b=json.dumps({'object':'list','data':[{'id':'fake-model','object':'model'}]}).encode(); self.send_response(200); self.send_header('Content-Length',str(len(b))); self.end_headers(); self.wfile.write(b)
  def do_POST(self):
    body=json.loads(self.rfile.read(int(self.headers.get('Content-Length','0')))); state.append(body); records.write_text(json.dumps(state,indent=2)+'\n')
    while not release.exists(): time.sleep(.02)
    msg={'role':'assistant','tool_calls':[{'index':0,'id':f'call_{len(state)}','type':'function','function':{'name':'communicate','arguments':json.dumps({'message':'released','end_turn':True,'output':{'message':'','data':{},'artifacts':[]}})}}]}
    b=('data: '+json.dumps({'id':f'fixture-{len(state)}','model':'fake-model','choices':[{'index':0,'delta':msg,'finish_reason':'tool_calls'}]})+'\n\ndata: [DONE]\n\n').encode(); self.send_response(200); self.send_header('Content-Type','text/event-stream'); self.send_header('Content-Length',str(len(b))); self.end_headers(); self.wfile.write(b)
ThreadingHTTPServer(('127.0.0.1',port),H).serve_forever()
