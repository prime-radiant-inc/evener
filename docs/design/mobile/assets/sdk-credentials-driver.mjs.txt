import assert from "node:assert/strict";
import { readFile, writeFile, mkdir, unlink } from "node:fs/promises";
import { createHash } from "node:crypto";
import { execFileSync } from "node:child_process";
import { fileURLToPath } from "node:url";
import { clientFromEnvironment } from "./node_modules/@evener/appwire-client/examples/connection.mjs";
import { runCredentials, safeCredentialSummary } from "./node_modules/@evener/appwire-client/examples/credentials-logic.mjs";
const root = fileURLToPath(new URL("../", import.meta.url));
const runtime = JSON.parse(await readFile(`${root}/runtime.json`, "utf8"));
process.env.EVENER_RPC_URL = runtime.url;
process.env.EVENER_TOKEN_FILE = `${root}/state/evener/auth-token`;
process.env.EVENER_CWD = root;
process.env.EVENER_CREDENTIAL_MUTATION = "1";
process.env.EVENER_CREDENTIAL_OWNED_HUB = runtime.url;
const { hub } = clientFromEnvironment();
const { hub: observer } = clientFromEnvironment();
const counts = {};
const removed = [];
const notifications = [];
const request = hub.request.bind(hub);
hub.request = async (method, params, options) => {
  counts[method] = (counts[method] ?? 0) + 1;
  const response = await request(method, params, options);
  if (method === "evener/auth/logout") removed.push(response.removed);
  return response;
};
observer.onNotification((notification) => {
  if (notification.method !== "evener/auth/updated") return;
  const params = notification.params;
  if (params.provider?.startsWith("sdk-credential-")) notifications.push({ provider: params.provider, activeSource: params.activeSource });
});
const created = [];
const checks = [];
const check = (name, condition) => { assert.ok(condition, name); checks.push(name); };
const stored = (value) => value.hasStoredFile ?? false;
const read = async (provider) => (await runCredentials(hub, { action: "status", params: { provider } })).readback;
const mutate = async (action, provider, value, reviewed) => {
  const result = await runCredentials(hub, { action, params: { provider, reviewed: reviewed ?? await read(provider), ...(value === undefined ? {} : { value }) }, ownedHub: runtime.url });
  assert.equal(result.execution, "unverified");
  assert.deepEqual(await observer.request("evener/auth/status", { provider }), result.readback);
  return result;
};
const key = "sdk-credential-key", env = "sdk-credential-env", adc = "sdk-credential-json", oauth = "sdk-credential-oauth";
const fakeJSON = JSON.stringify({ type: "authorized_user", client_id: "owned-fixture-client", client_secret: "owned-fixture-client-secret", refresh_token: "owned-fixture-refresh" });
const adcDir = `${root}/home/.config/gcloud`;
const adcFile = `${adcDir}/application_default_credentials.json`;
await mkdir(adcDir, { recursive: true });
await writeFile(adcFile, fakeJSON, { flag: "wx", mode: 0o600 });
try {
  await Promise.all([hub.connect(), observer.connect()]);
  for (const params of [
    { name: key, base: "anthropic", baseUrl: "http://127.0.0.1:1" },
    { name: env, base: "anthropic", baseUrl: "http://127.0.0.1:1", apiKeyEnv: "CREDENTIAL_FIXTURE_KEY" },
    { name: adc, base: "google-vertex", vars: { GOOGLE_VERTEX_PROJECT: "owned-fixture", GOOGLE_VERTEX_LOCATION: "global" } },
    { name: oauth, base: "openai-codex" },
  ]) { await hub.request("evener/instance/create", params); created.push(params.name); }
  const listed = await runCredentials(hub);
  check("four owned instances appear exactly once in credential list", [key,env,adc,oauth].every(provider => listed.readback.filter(row => row.provider === provider).length === 1));
  const keyBefore = await read(key);
  check("key instance starts without a credential", !keyBefore.signedIn && !stored(keyBefore));
  const set = await mutate("apiKey/set", key, "owned-fixture-key-value");
  check("stored API key acknowledged and visible to independent client", set.outcome === "acknowledged" && stored(set.readback) && set.readback.activeSource === "store");
  const beforeRejected = counts["evener/auth/apiKey/clear"] ?? 0;
  await assert.rejects(mutate("apiKey/clear", key, undefined, keyBefore));
  check("stale reviewed status rejected before dispatch", (counts["evener/auth/apiKey/clear"] ?? 0) === beforeRejected && stored(await read(key)));
  const clear = await mutate("apiKey/clear", key);
  check("cleared API key no longer authenticates", clear.outcome === "acknowledged" && !stored(clear.readback) && !clear.readback.signedIn);
  await mutate("logout", key);
  check("absent logout reports no stored removal", removed.at(-1) === false);
  check("environment credential resolves", (await read(env)).activeSource === "env:CREDENTIAL_FIXTURE_KEY");
  await mutate("logout", env);
  check("environment-only logout reports no stored removal", removed.at(-1) === false && (await read(env)).signedIn);
  const setEnv = await mutate("apiKey/set", env, "owned-fixture-shadow-key");
  check("stored key takes precedence over environment", stored(setEnv.readback) && setEnv.readback.activeSource === "store");
  const logoutEnv = await mutate("logout", env);
  check("logout clears stored key and reveals environment", removed.at(-1) === true && !stored(logoutEnv.readback) && logoutEnv.readback.activeSource === "env:CREDENTIAL_FIXTURE_KEY");
  const adcBefore = await read(adc);
  check("ADC file resolves without a stored credential", adcBefore.activeSource === "adc" && !stored(adcBefore));
  const invalidJSON = await mutate("credentialJson/set", adc, "{}");
  check("server rejects incomplete Google JSON without changing state", invalidJSON.outcome === "uncertain" && !stored(invalidJSON.readback) && invalidJSON.readback.activeSource === "adc");
  const setJSON = await mutate("credentialJson/set", adc, fakeJSON);
  check("Google credential JSON stores successfully", setJSON.outcome === "acknowledged" && stored(setJSON.readback) && setJSON.readback.activeSource === "store");
  const clearJSON = await mutate("apiKey/clear", adc);
  check("clearing stored JSON preserves ADC", clearJSON.outcome === "acknowledged" && !stored(clearJSON.readback) && clearJSON.readback.activeSource === "adc");
  await mutate("logout", adc);
  check("ADC-only logout reports no stored removal", removed.at(-1) === false && (await read(adc)).activeSource === "adc");
  const oauthBefore = counts["evener/auth/apiKey/set"];
  await assert.rejects(mutate("apiKey/set", oauth, "owned-fixture-refused-key"));
  check("OAuth instance rejects key recipe before dispatch", counts["evener/auth/apiKey/set"] === oauthBefore && !stored(await read(oauth)));
  await mutate("apiKey/set", key, "owned-fixture-cli-key");
  const paramsFile = `${root}/credential-params.json`;
  await writeFile(paramsFile, JSON.stringify({ provider:key,reviewed:await read(key) }), { flag:"wx",mode:0o600 });
  const cliPath = fileURLToPath(new URL("./node_modules/@evener/appwire-client/examples/credentials.mjs", import.meta.url));
  const output = execFileSync(process.execPath, [cliPath], { env:{ ...process.env,EVENER_CREDENTIAL_ACTION:"apiKey/clear",EVENER_CREDENTIAL_PARAMS_FILE:paramsFile },encoding:"utf8",stdio:["ignore","pipe","pipe"] });
  assert.deepEqual(JSON.parse(output), { action:"apiKey/clear",outcome:"acknowledged",execution:"unverified",supported:true,signedIn:false });
  check("packaged CLI clears credential with only allowlisted summary", !stored(await read(key)));
  await unlink(paramsFile);
  const summary = safeCredentialSummary(await runCredentials(hub));
  assert.deepEqual(Object.keys(summary).sort(),["action","execution","outcome","providerCount"]);
  check("list summary excludes credential metadata", summary.providerCount >= 4);
  check("independent client observes all eleven successful credential updates", notifications.length === 11);
} finally {
  for (const name of created.reverse()) await hub.request("evener/instance/remove", {name});
  const final = await hub.request("evener/instance/list", {});
  const rows = final.instances;
  assert.ok(Array.isArray(rows), "instance list response has entries");
  check("all owned instances removed", rows.every(row => ![key,env,adc,oauth].includes(row.name)));
  check("ADC fixture file unchanged", await readFile(adcFile,"utf8") === fakeJSON);
  await unlink(adcFile);
  hub.close(); observer.close();
}
const receipt = { sourceHead:runtime.sourceHead,binarySha256:runtime.binarySha256,packageSha256:runtime.packageSha256,driverSha256:createHash("sha256").update(await readFile(fileURLToPath(import.meta.url))).digest("hex"), checks,requestCounts:counts,logoutRemoved:removed,authNotificationCount:notifications.length,providerExecution:"unverified",credentialInputs:"fixture values only",transport:"direct authenticated loopback hub" };
await writeFile(`${root}/receipt.json`,JSON.stringify(receipt,null,2)+"\n");
console.log(JSON.stringify(receipt,null,2));
