import assert from 'node:assert/strict';
import {readFile,writeFile,stat} from 'node:fs/promises';
import {createHash} from 'node:crypto';
import {join} from 'node:path';
import {AppwireClient} from '/tmp/evener-notifications-consumer.yrcaOf/node_modules/@evener/appwire-client/dist/index.js';
import WebSocket from '/tmp/evener-notifications-consumer.yrcaOf/node_modules/ws/index.js';
const root=process.argv[2]??'.'; const checkpoint=process.argv[3]??'baseline'; const labels=process.argv.slice(4); assert.match(checkpoint,/^[a-z-]+$/); assert(labels.length>0&&labels.every(x=>x==='A'||x==='B'));
const userMarker=process.env.EVENER_EXPECTED_A_USER_MARKER??'owned ipad a held turn 829c';
const agentMarker=process.env.EVENER_EXPECTED_A_AGENT_MARKER??'fixture-A-held-1';
const receipt={checkpoint,hubs:{}}; const files=[];
async function privateRead(path){return JSON.parse(await readFile(path,'utf8'));}
async function save(name,value){const path=join(root,'run',`readback-${checkpoint}-${name}.json`);await writeFile(path,JSON.stringify(value,null,2)+'\n',{flag:'wx',mode:0o600});files.push({path,sha256:createHash('sha256').update(await readFile(path)).digest('hex'),mode:(await stat(path)).mode&0o777});}
for(const label of labels){const rt=await privateRead(join(root,label,'runtime.json'));const binding=await privateRead(join(root,label,'binding.json'));const baseline=await privateRead(join(root,label,'baseline.json'));const token=(await readFile(rt.tokenFile,'utf8')).trim();const hub=new AppwireClient({url:rt.rpcURL,clientInfo:{name:'ipad-two-hub-readback',version:'1'},socketFactory:u=>new WebSocket(u,{headers:{Authorization:`Bearer ${token}`}})});try{await hub.connect();const read=await hub.request('thread/read',{ref:binding.ref,includeTurns:true,itemLimit:40,itemsView:'full',subscribe:false});assert.equal(read.thread.evener.ref,binding.ref);assert.equal(read.thread.evener.instanceId,binding.instanceId);const items=read.thread.turns.flatMap(t=>t.items??[]);const users=items.filter(i=>i.type==='userMessage').map(i=>i.text);const agents=items.filter(i=>i.type==='agentMessage').map(i=>i.text);if(checkpoint==='after-held'&&label==='A'){assert(users.includes(userMarker),`missing A marker in ${label}`);assert(agents.includes(agentMarker),`missing A response in ${label}`);}if(checkpoint==='after-held'&&label==='B'){const baseItems=baseline.thread.turns.flatMap(t=>t.items??[]);assert.deepEqual(users,baseItems.filter(i=>i.type==='userMessage').map(i=>i.text));assert.deepEqual(agents,baseItems.filter(i=>i.type==='agentMessage').map(i=>i.text));}await save(label,{runtime:{hubPort:rt.hubPort,rpcURL:rt.rpcURL},binding,read,users,agents});receipt.hubs[label]={hubPort:rt.hubPort,ref:binding.ref,instanceId:binding.instanceId,turns:read.thread.turns.length,userMessageCount:users.length,agentMessageCount:agents.length};}finally{hub.close();}}
await save('receipt',{checkpoint,hubs:receipt.hubs,files});console.log(JSON.stringify({status:'passed',checkpoint,hubs:Object.fromEntries(Object.entries(receipt.hubs).map(([k,v])=>[k,{ref:v.ref,turns:v.turns,userMessageCount:v.userMessageCount,agentMessageCount:v.agentMessageCount}]))}));
