const root = realpathSync(process.env.SAVED_ITEMS_ROOT ?? process.cwd());

const token = readFileSync(`${root}/state/evener/auth-token`, 'utf8').trim();
const rpcUrl = `ws://127.0.0.1:${runtime.hubPort}/rpc`;
const makeClient = () => new AppwireClient({ url: rpcUrl, clientInfo: { name: 'saved-items-qualification', version: '1' }, socketFactory: endpoint => new WebSocket(endpoint, { headers: { Authorization: `Bearer ${token}` } }) });
const records = [];
const privateFile = (name, value) => { const path = `${root}/run/qualification-v3/${name}.json`; writeFileSync(path, `${JSON.stringify(value)}\n`, { flag: 'wx', mode: 0o600 }); records.push({ name, sha256: createHash('sha256').update(readFileSync(path)).digest('hex'), mode: (statSync(path).mode & 0o777).toString(8) }); return path; };
async function readCatalog(catalog) { const client = makeClient(); await client.connect(); const result = await client.request('evener/navigation/read', { representationVersion: 2, resource: 'catalog', catalog, limit: 100, offset: 0 }); client.close(); return result; }
async function readLocation(ref) { const client = makeClient(); await client.connect(); const result = await client.request('evener/navigation/read', { representationVersion: 2, resource: 'location', ref }); client.close(); return result; }
async function run(action, params, name) { const client = makeClient(); const result = await runSavedItems(client, { action, params, ownedHub: rpcUrl }); client.close(); assert.equal(result.outcome, 'acknowledged', `${name}: ${result.outcome}`); assert.ok(result.readback.navigation); if (action.endsWith('Delete')) { assert.ok(Array.isArray(result.readback.deleted)); assert.ok(Array.isArray(result.readback.skipped)); } const safe={action:result.action,outcome:result.outcome,execution:result.execution,readback: result.readback ? {ok:result.readback.ok??null,deleted:result.readback.deleted??[],skipped:result.readback.skipped??[],navigation:result.readback.navigation??null} : null}; privateFile(`${name}-readback`,safe); records.push({ name, action, deleted: result.readback.deleted?.length ?? null, skipped: result.readback.skipped?.length ?? null, navigationTargets: result.readback.navigation.targets.length }); return result; }
async function waitTurn(client, ref, turnId) { for (let i = 0; i < 1200; i++) { const result = await client.request('thread/read', { ref, includeTurns: true, subscribe: false }); if (['idle', 'awaiting'].includes(result.thread.status.type) && result.thread.turns.some(turn => turn.id === turnId && turn.status === 'completed')) return result; await new Promise(resolve => setTimeout(resolve, 50)); } throw new Error('turn completion timeout'); }
async function waitNotLoaded(client, ref) { for (let i = 0; i < 1200; i++) { const result = await client.request('thread/read', { ref, includeTurns: false, subscribe: false }); if (result.thread.status.type === 'notLoaded') { const nav = await client.request('evener/navigation/read',{representationVersion:2,resource:'location',ref}); const entity = nav.data.entities.find(e=>e.value?.ref===ref); if(entity && entity.value.live !== true) return result; } await new Promise(resolve => setTimeout(resolve, 50)); } throw new Error('notLoaded timeout'); }
export async function qualify({ projectKey, projectDir, sessionRef, sessionId, throwawayKey, throwawayDir, throwawayRef }) {
  process.env.EVENER_SAVED_ITEMS_MUTATION = '1'; process.env.EVENER_RPC_URL = rpcUrl;
  for (const [favorited, name] of [[true, 'favorite-on'], [false, 'favorite-off']]) { await run('favorite', { kind: 'project', id: projectKey, favorited, reviewed: { kind: 'project', id: projectKey } }, name); const read = await readCatalog('projects'); const entity = read.data.entities.find(item => item.value?.key === projectKey); assert.ok(entity, 'project exists'); assert.equal(entity.value.favorite ?? false, favorited); records.push({ name: `${name}-readback`, favorite: entity.value.favorite ?? false }); }
  for (const [archived, name] of [[true, 'project-archive-on'], [false, 'project-archive-off']]) { await run('archive', { kind: 'project', id: projectKey, workingDir: projectDir, archived, reviewed: { kind: 'project', id: projectKey, workingDir: projectDir } }, name); const active = await readCatalog('projects'); const archivedView = await readCatalog('archived_projects'); const projected = [...active.data.entities,...archivedView.data.entities].find(item => item.value?.key === projectKey); assert.ok(projected); assert.equal(projected.value.is_archived ?? false, archived); assert.equal(active.data.entities.some(item => item.value?.key === projectKey), !archived); assert.equal(archivedView.data.entities.some(item => item.value?.key === projectKey), archived); records.push({ name: `${name}-readback`, active: active.data.entities.some(item => item.value?.key === projectKey), archived: archivedView.data.entities.some(item => item.value?.key === projectKey) }); }
  for (const [archived, name] of [[true, 'session-archive-on'], [false, 'session-archive-off']]) { await run('archive', { kind: 'session', id: sessionId, archived, reviewed: { kind: 'session', id: sessionId } }, name); const location = await readLocation(sessionRef); assert.equal(location.data.metadata.tier, archived ? 'archived' : 'current'); assert.equal(location.data.entities[0].value.session_id, sessionId); records.push({ name: `${name}-readback`, tier: location.data.metadata.tier, sessionId: location.data.entities[0].value.session_id }); }
  const client = makeClient(); await client.connect(); await client.request('thread/shutdown', { ref: sessionRef }); await waitNotLoaded(client, sessionRef); client.close(); const deleted = await run('sessionDelete', { ref: sessionRef, reviewed: { ref: sessionRef }, confirmTarget: { ref: sessionRef } }, 'session-delete'); assert.deepEqual(deleted.readback.deleted,[sessionId]); assert.equal(deleted.readback.skipped.length, 0);
  const independent = makeClient(); await independent.connect(); await assert.rejects(() => independent.request('thread/read', { ref: sessionRef, includeTurns: false, subscribe: false }), error => { assert.equal(error.code,-32014); assert.equal(error.evenerErrorInfo,'sessionUnavailable'); records.push({name:'session-delete-independent-read', code:error.code}); return true; }); independent.close();
  const throwClient = makeClient(); await throwClient.connect(); await throwClient.request('thread/shutdown', { ref: throwawayRef }); await waitNotLoaded(throwClient, throwawayRef); throwClient.close(); const projectDeleted = await run('projectDelete', { key: throwawayKey, workingDir: throwawayDir, reviewed: { key: throwawayKey, workingDir: throwawayDir }, confirmTarget: { key: throwawayKey, workingDir: throwawayDir } }, 'project-delete'); assert.deepEqual(projectDeleted.readback.deleted,[throwawayRef.slice('local:'.length)]); assert.equal(projectDeleted.readback.skipped.length, 0); const remaining = await readCatalog('projects'); const archivedRemaining = await readCatalog('archived_projects'); assert(![...remaining.data.entities, ...archivedRemaining.data.entities].some(item => item.value?.key === throwawayKey));
  const receipt = { status: 'passed', runtime: { hubPID: runtime.hubPID, providerPID: runtime.providerPID, hubPort: runtime.hubPort, providerPort: runtime.providerPort }, projectKey, sessionRef, sessionId, throwawayKey, actions: records, privateOutputs: records.filter(record => record.mode), limitations: ['scripted fixture provider', 'no UI or public provider'] }; privateFile('saved-items-receipt', receipt); return receipt;
}
