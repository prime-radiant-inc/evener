set -euo pipefail
ROOT=/tmp/evener-upgrade-qualify.$$
mkdir -m 700 "$ROOT"
PKG=/tmp/evener-upgrade-consumer.fc81fe/node_modules/@evener/appwire-client/examples
CONTROL=http://127.0.0.1:51243
RPC=ws://127.0.0.1:51242/rpc
TOKEN=/var/folders/46/dz2z92w907j150sqxn8b8y1c0000gn/T/TestNativeUpgradeHarness74725591/001/hub-state/auth-token
printf '%s' '{}' >"$ROOT/review-params.json"
EVENER_RPC_URL=$RPC EVENER_TOKEN_FILE=$TOKEN EVENER_CWD=/tmp EVENER_HUB_UPGRADE_ACTION=review EVENER_HUB_UPGRADE_PARAMS_FILE="$ROOT/review-params.json" EVENER_HUB_UPGRADE_OUTPUT_FILE="$ROOT/review-output.json" node "$PKG/hub-upgrade.mjs" >"$ROOT/review.stdout" 2>"$ROOT/review.stderr"

cat "$ROOT/retry.stdout"
cat "$ROOT/retry-output.json"
curl -fsS "$CONTROL/status" >"$ROOT/status-retry.json"
