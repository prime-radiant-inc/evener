function waitForEvent(method,ref,start,turnId) {
  return new Promise((resolve,reject)=>{
    const timer=setTimeout(()=>{pending.delete(check);reject(new Error(`No ${method} completion event`));},60000);
    function check(){
      const event=events.slice(start).find(e=>e.method===method && e.params.ref===ref && (turnId===undefined || e.params.turnId===turnId));
      if(event){clearTimeout(timer);pending.delete(check);resolve(event);}
    }
    pending.add(check);check();
  });
}
async function waitFor(condition,label){

}
