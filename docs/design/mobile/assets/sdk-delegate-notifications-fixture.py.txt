import hashlib
import json
import os
import pathlib
import signal
import socket
import subprocess
import threading
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
from provider_state import choose_stage

root = pathlib.Path(__file__).resolve().parent
records = []
lock = threading.Lock()
stop = threading.Event()
summary_marker = 'fixture-summary-delegate-6d1b'
qualification_marker = 'delegate-notification-qualification-6d1b'
parent_delegate_sent = False
child_result_sent = False
parent_result_sent = False
task_id_ready = threading.Event()
qualification_task_id = None
child_result_ready = threading.Event()

class Handler(BaseHTTPRequestHandler):
    def log_message(self, *_):
        pass

    def send_json(self, value):
        data = json.dumps(value).encode()
        self.send_response(200)
        self.send_header('Content-Type', 'application/json')
        self.send_header('Content-Length', str(len(data)))
        self.end_headers()
        self.wfile.write(data)

    def do_GET(self):
        if self.path == '/v1/models':
            self.send_json({'object':'list','data':[{'id':'fixture-model','object':'model'}]})
        elif self.path == '/status':
            with lock:
                self.send_json({'requests':list(records)})
        else:
            self.send_error(404)

    def do_POST(self):
        global qualification_task_id
        if self.path == '/shutdown':
            self.send_json({'stopping':True})
            stop.set()
            return
        if self.path != '/v1/chat/completions':
            self.send_error(404)
            return
        global parent_delegate_sent, child_result_sent, parent_result_sent
        body = json.loads(self.rfile.read(int(self.headers['Content-Length'])))
        tools = body.get('tools', [])
        messages = body.get('messages', [])
        serialized = json.dumps(messages, sort_keys=True)
        qualifying = qualification_marker in serialized
        offered = {t.get('function', {}).get('name'): t.get('function', {}) for t in tools}
        names = sorted(offered)
        response_schema = body.get('response_format', {}).get('json_schema', {}).get('schema', {})
        schema_props = sorted(response_schema.get('properties', {}))
        kind = 'turn' if tools else ('name' if schema_props == ['name'] else 'summary')
        with lock:
            index = len(records) + 1
            records.append({'index': index, 'kind': kind, 'qualification': qualifying,
                            'toolNames': names, 'messageCount': len(messages),
                            'bodyHash': hashlib.sha256(json.dumps(body, sort_keys=True).encode()).hexdigest()})
            (root/'provider-requests.pending').write_text(json.dumps(records, indent=2)+'\n')
            (root/'provider-requests.pending').replace(root/'provider-requests.json')
            with (root/'provider-request-bodies.jsonl').open('a') as raw:
                raw.write(json.dumps(body, sort_keys=True)+'\n')
        finish = 'stop'
        message = {'role': 'assistant'}
        if kind == 'turn':
            assert 'communicate' in offered, 'result tool communicate was not offered'
            stage = choose_stage(qualifying, 'delegate' in offered, parent_delegate_sent, child_result_sent)
            function_name = 'communicate'
            arguments = {'message': 'delegate fixture complete', 'end_turn': True,
                         'output': {'message': '', 'data': {}, 'artifacts': []}}
            if stage == 'parent-delegate':
                parent_delegate_sent = True
                function_name = 'delegate'
                props = offered['delegate'].get('parameters', {}).get('properties', {})
                arguments = {'prompt': 'Complete the bounded delegate fixture check for delegate-notification-qualification-6d1b and report delegate-child-complete.'}
                if 'agent_type' in props and props['agent_type'].get('enum'):
                    arguments['agent_type'] = props['agent_type']['enum'][0]
                if 'delegation_allowance' in props:
                    arguments['delegation_allowance'] = 0
            elif stage == 'child-result':
                child_result_sent = True
                child_result_ready.set()
                arguments['message'] = 'delegate-child-complete'
            elif stage == 'parent-wait-child':
                if not child_result_ready.wait(timeout=30):
                    raise RuntimeError('child delegate result was not delivered before parent completion')
                child_result_sent = True
            elif stage == 'parent-result':
                parent_result_sent = True
                arguments['message'] = 'delegate-parent-complete'
            elif qualifying:
                raise RuntimeError('unexpected additional delegate qualification request')
            with lock:
                records[index-1]['authoredTool'] = function_name
                records[index-1]['stage'] = stage
                (root/'provider-requests.pending').write_text(json.dumps(records, indent=2)+'\n')
                (root/'provider-requests.pending').replace(root/'provider-requests.json')
            message['tool_calls'] = [{'index': 0, 'id': f'call_fixture_{index}', 'type': 'function',
                'function': {'name': function_name, 'arguments': json.dumps(arguments)}}]
            finish = 'tool_calls'
        else:
            message['content'] = json.dumps({'name': 'Delegate fixture'}) if kind == 'name' else summary_marker
        base = {'id':f'fixture-{index}','model':'fixture-model'}
        usage = {'prompt_tokens':100,'completion_tokens':10,'total_tokens':110}
        if body.get('stream'):
            self.send_response(200)
            self.send_header('Content-Type','text/event-stream')
            self.end_headers()
            chunks = [{**base,'choices':[{'index':0,'delta':message,'finish_reason':None}]},
                      {**base,'choices':[{'index':0,'delta':{},'finish_reason':finish}],'usage':usage}]
            for chunk in chunks:
                self.wfile.write(('data: '+json.dumps(chunk)+'\n\n').encode())
                self.wfile.flush()
            self.wfile.write(b'data: [DONE]\n\n')
        else:
            self.send_json({**base,'choices':[{'index':0,'message':message,'finish_reason':finish}],'usage':usage})

server = ThreadingHTTPServer(('127.0.0.1',0), Handler)
threading.Thread(target=server.serve_forever,daemon=True).start()
provider_port=server.server_port
with socket.socket() as probe:
    probe.bind(('127.0.0.1',0))
    hub_port=probe.getsockname()[1]
providers=root/'config/evener/providers.toml'
providers.write_text(f'''schema = 2
default = "fixture"
[providers.fixture]
base = "openai-compatible"
base_url = "http://127.0.0.1:{provider_port}/v1"
auth = "none"
protocol = "openai-chat"
cheap_model = "fixture-model"
''')
config=root/'config/evener/hub.toml'
config.write_text(f'addr = "127.0.0.1:{hub_port}"\nplugin_auto_upgrade = false\n')
env={'PATH':'/opt/homebrew/bin:/usr/bin:/bin:/usr/sbin:/sbin','USER':'jesse','LANG':'en_US.UTF-8',
     'TMPDIR':str(root/'run'),
     'XDG_CONFIG_HOME':str(root/'config'),'XDG_STATE_HOME':str(root/'state'),
     'XDG_CACHE_HOME':str(root/'cache'),'XDG_DATA_HOME':str(root/'data'),'XDG_RUNTIME_DIR':str(root/'run'),
     'EVENER_OFFLINE':'1','EVENER_PROVIDERS_CONFIG':str(providers),'EVENER_CREDENTIALS_CONFIG':str(root/'config/evener/credentials.toml')}
binary=str(root/'bin/evener')
with (root/'hub.log').open('xb') as log:
    os.chmod(root/'hub.log',0o600)
    process=subprocess.Popen([binary,'hub','--config',str(config),'--evener',binary],cwd=root/'workspace',env=env,stdout=log,stderr=log,start_new_session=True)
record={'root':str(root),'providerPID':os.getpid(),'hubPID':process.pid,'hubPort':hub_port,'providerPort':provider_port,'environment':env}
(root/'runtime.json').write_text(json.dumps(record,indent=2)+'\n')
print(json.dumps({k:v for k,v in record.items() if k != 'environment'}),flush=True)
signal.signal(signal.SIGTERM,lambda *_:stop.set())
signal.signal(signal.SIGINT,lambda *_:stop.set())
try:
    stop.wait()
finally:
    process.terminate()
    try: process.wait(timeout=20)
    except subprocess.TimeoutExpired:
        process.kill()
        process.wait()
    server.shutdown()
    server.server_close()
    token=root/'state/evener/auth-token'
    if token.exists():token.unlink()
    print(json.dumps({'hubExit':process.returncode,'credentialRemoved':not token.exists()}),flush=True)
