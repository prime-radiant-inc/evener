const read=(name)=>fs.readFile(root+'/'+name+'.json','utf8').then(JSON.parse);
function observe(ref){
 const seen=new Map(), waiters=new Set();
 const stop=hub.onNotification(event=>{
  if(event.params.ref!==ref || event.method!=='turn/completed')return;
  seen.set(event.params.turn.id,event.params.turn); for(const check of waiters)check();
 });
 return {stop,completed:turnId=>new Promise((resolve,reject)=>{
  const timer=setTimeout(()=>{waiters.delete(check);reject(new Error('completion missing'))},30000);
  function check(){if(seen.has(turnId)){clearTimeout(timer);waiters.delete(check);resolve(seen.get(turnId))}}
  waiters.add(check);check();
 })};
}
let stage='connect';
try{
 await hub.connect();

  const old=await read('binding');
  const observed=observe(old.ref);
  try{
   await hub.request('thread/read',{ref:old.ref,includeTurns:false,subscribe:true});
   for(let round=0;round<2;round++){
    const reviewed=await runQuestions(hub,{params:{ref:old.ref}});
    const selections=round===0?[{resolution:{kind:'option',labels:['Apply now']},note:config.note},{resolution:{kind:'option',labels:['Review locally','Share for review']},note:''}]:[{resolution:{kind:'option',labels:['Continue verification']},note:''}];

    assert.equal((await observed.completed(answer.receipt.turnId)).status,'completed');
   }
   await hub.request('thread/shutdown',{ref:old.ref});
   await save('excluded-setup',{...old,reason:'Question was already pending before the ordinary draft could be entered.',completed:true,shutdown:true});
  }finally{observed.stop();await hub.request('thread/unsubscribe',{ref:old.ref})}

  const binding={ref:thread.evener.ref,instanceId:thread.evener.instanceId,threadId:thread.id};
  assert.ok(binding.instanceId);await save('binding',binding);
