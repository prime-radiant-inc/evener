import { execFileSync } from "node:child_process";
import { createHash } from "node:crypto";
import { chmod, mkdir, readFile, stat, writeFile } from "node:fs/promises";
import { join, resolve } from "node:path";

const args = Object.fromEntries(process.argv.slice(2).map((arg) => {
  const [key, ...value] = arg.slice(2).split("=");
  return [key, value.join("=")];
}));
for (const key of ["rpc", "token-file", "cwd", "params-file", "package-root", "tarball", "output-dir"])
  if (!args[key]) throw new Error(`Missing --${key}=...`);
const outputDir = resolve(args["output-dir"]);
await mkdir(outputDir, { recursive: true });
const tokenFile = resolve(args["token-file"]);
const packageRoot = resolve(args["package-root"]);
const cli = join(packageRoot, "examples", "maintenance-checks.mjs");
const common = {
  EVENER_RPC_URL: args.rpc,
  EVENER_TOKEN_FILE: tokenFile,
  EVENER_CWD: resolve(args.cwd),
  EVENER_MAINTENANCE_MUTATION: "1",
  EVENER_MAINTENANCE_OWNED_HUB: args.rpc,
};
const cases = [
  ["ping", {}],
  ["auth", { EVENER_MAINTENANCE_ACTION: "auth/test", EVENER_MAINTENANCE_PARAMS_FILE: resolve(args["params-file"]) }],
  ["plugin", { EVENER_MAINTENANCE_ACTION: "plugin/checkNow" }],
];
for (const [name, extra] of cases) {
  const output = join(outputDir, `${name}.json`);
  const stdout = execFileSync(process.execPath, [cli], {
    env: { ...process.env, ...common, ...extra, EVENER_MAINTENANCE_OUTPUT_FILE: output },
    encoding: "utf8",
  });
  await writeFile(join(outputDir, `${name}.stdout`), stdout, { mode: 0o600 });
}
const hash = async (path) => ({
  path,
  sha256: createHash("sha256").update(await readFile(path)).digest("hex"),
  mode: (await stat(path)).mode & 0o777,
});
const receipt = {
  rpc: args.rpc,
  package: await hash(resolve(args.tarball)),
  outputs: Object.fromEntries(await Promise.all(cases.map(async ([name]) => [name, {
    private: await hash(join(outputDir, `${name}.json`)),
    stdout: await hash(join(outputDir, `${name}.stdout`)),
  }]))),
  sourceInputs: args["source-inputs"] ? JSON.parse(await readFile(resolve(args["source-inputs"]), "utf8")) : undefined,
};
await writeFile(join(outputDir, "maintenance-receipt.json"), `${JSON.stringify(receipt, null, 2)}\n`, { mode: 0o600 });
await chmod(join(outputDir, "maintenance-receipt.json"), 0o600);
console.log(JSON.stringify({ outputDir, operations: cases.map(([name]) => name) }));
