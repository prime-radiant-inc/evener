import assert from 'node:assert/strict';
import {readFileSync,writeFileSync} from 'node:fs';
import {AppwireClient} from '/tmp/evener-notifications-consumer.yrcaOf/node_modules/@evener/appwire-client/dist/index.js';
import WebSocket from '/tmp/evener-notifications-consumer.yrcaOf/node_modules/ws/index.js';
import {runSessionNotifications} from '/tmp/evener-notifications-consumer.yrcaOf/node_modules/@evener/appwire-client/examples/session-notifications-logic.mjs';
const rt=JSON.parse(readFileSync('runtime.json')),session=JSON.parse(readFileSync('session.json')),token=readFileSync('state/evener/auth-token','utf8').trim();
const make=()=>new AppwireClient({url:`ws://127.0.0.1:${rt.hubPort}/rpc`,clientInfo:{name:'bounded-event-overflow',version:'1'},socketFactory:url=>new WebSocket(url,{headers:{Authorization:`Bearer ${token}`}})});
const actor=make(),observer=make();const events=[],waiters=new Set();const off=observer.onNotification(n=>{events.push(n);for(const check of waiters)check();});
function untilName(name){return new Promise((resolve,reject)=>{const timer=setTimeout(()=>{waiters.delete(check);reject(new Error('name notification timeout'));},15000);function check(){if(events.some(n=>n.method==='evener/thread/name/changed'&&n.params.ref===session.ref&&n.params.name===name)){clearTimeout(timer);waiters.delete(check);resolve();}}waiters.add(check);check();});}
try{await actor.connect();const original=(await actor.request('thread/read',{ref:session.ref,includeTurns:false,subscribe:false})).thread.name;const result=await runSessionNotifications(observer,{ref:session.ref,maxEvents:1,observe:async()=>{for(const name of ['Bounded overflow fixture',original]){await actor.request('evener/thread/name/set',{ref:session.ref,name});await untilName(name);}}});assert.equal(result.outcome,'uncertain');assert.equal(result.overflow,true);assert.equal(result.events.length,1);assert.equal(result.eventCount,2);assert.equal(result.readback.thread.name,original);assert.equal((await actor.request('thread/read',{ref:session.ref,includeTurns:false,subscribe:false})).thread.name,original);writeFileSync('run/overflow-receipt.json',JSON.stringify({status:'passed',scope:'Actual two rename events overflow retention limit of one; current name restored and independently read.',sourceCommit:'58d1b079f',result})+'\n',{flag:'wx',mode:0o600});console.log(JSON.stringify({status:'passed',outcome:result.outcome,retained:result.events.length,eventCount:result.eventCount}));}finally{off();observer.close();actor.close();}
