import hashlib
import json
import os
import pathlib
import subprocess
import urllib.parse

root=pathlib.Path(__file__).resolve().parent
fixture=pathlib.Path('/var/folders/46/dz2z92w907j150sqxn8b8y1c0000gn/T/evener-compaction-effect-02oikf0o')
runtime=json.loads((fixture/'runtime.json').read_text())
url=f"ws://127.0.0.1:{runtime['hubPort']}/rpc"
output_dir=root/'setup-results'
output_dir.mkdir(mode=0o700)
entry=root/'node_modules/@evener/appwire-client/examples/hub-setup.mjs'
env={'PATH':os.environ['PATH'],'LANG':'en_US.UTF-8','NODE_DISABLE_COMPILE_CACHE':'1','EVENER_RPC_URL':url,
     'EVENER_TOKEN_FILE':str(fixture/'state/evener/auth-token'),'EVENER_CWD':str(fixture/'workspace'),
     'EVENER_HUB_SETUP_MUTATION':'1','EVENER_HUB_SETUP_OWNED_HUB':url}
target=fixture/'workspace/sdk-created/child'
records=[]
def artifact(path):
    return {'path':str(path),'sha256':hashlib.sha256(path.read_bytes()).hexdigest(),'mode':oct(path.stat().st_mode & 0o777)}
for name,action,params in [('create','dirs',{'path':str(target)}),('existing','dirs',{'path':str(target)}),('pairing','pairing',{'origin':'https://hub.fixture.example'})]:
    params_file=output_dir/(name+'-params.json')
    with params_file.open('x') as file:
        os.chmod(params_file,0o600)
        json.dump(params,file)
    output=output_dir/(name+'.json')
    completed=subprocess.run(['node',str(entry)],env={**env,'EVENER_HUB_SETUP_ACTION':action,'EVENER_HUB_SETUP_PARAMS_FILE':str(params_file),'EVENER_HUB_SETUP_OUTPUT_FILE':str(output)},capture_output=True,text=True)
    assert completed.returncode==0, f'{name} CLI did not succeed'
    stdout=json.loads(completed.stdout)
    response=json.loads(output.read_text())
    assert output.stat().st_mode & 0o777 == 0o600
    assert completed.stderr==''
    assert stdout['action']==action and stdout['outputWritten'] is True
    if action=='dirs':
        assert target.is_dir()
        assert pathlib.Path(response['path']).resolve()==target.resolve()
        assert response['created'] is (name=='create')
        assert stdout=={'action':'dirs','outcome':'acknowledged','execution':'unverified','created':name=='create','outputWritten':True}
    else:
        parsed=urllib.parse.urlsplit(response['authUrl'])
        assert parsed.scheme=='https' and parsed.netloc=='hub.fixture.example'
        assert urllib.parse.unquote(parsed.path.removeprefix('/auth/'))==(fixture/'state/evener/auth-token').read_text().strip()
        assert not parsed.query and not parsed.fragment
        assert stdout=={'action':'pairing','outcome':'read','execution':'unverified','outputWritten':True}
    records.append({'action':name,'stdout':stdout,'params':params,'output':artifact(output)})
    if action=='pairing':
        output.unlink()
        records[-1]['credentialOutputRemoved']=True
receipt={'sdk':json.loads((root/'inputs.json').read_text()),'runtime':json.loads((fixture/'inputs.json').read_text()),'recipeCommit':'a229b0c82','operations':records,'driver':artifact(pathlib.Path(__file__).resolve()),'limitations':['Pairing generation only: no phone import, DNS/reachability or LAN qualification.','Directory creation qualified only inside the disposable fixture workspace.']}
(root/'setup-receipt.json').write_text(json.dumps(receipt,indent=2)+'\n')
print(json.dumps({'actions':[r['action'] for r in records],'directoryExists':target.is_dir(),'pairingCredentialOutputRemoved':True}))
