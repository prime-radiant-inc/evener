#include <stdio.h>
#include <stdlib.h>
#include <string.h>

static void tok(const char*in,char*w[][64],int*n){
  static char b[4096];int i=0,j=0;*n=0;for(;in[i]&&j<4000;i++){
    unsigned char c=in[i];
    if(isalnum(c)||c=='\''){b[j++]=c;}
    else{if(j&&*n<128){b[j]=0;strcpy((char*)w[(*n)++],b);j=0;}if(!isspace(c)&&*n<128){b[0]=c;b[1]=0;strcpy((char*)w[(*n)++],b);}}
  }if(j&&*n<128){b[j]=0;strcpy((char*)w[(*n)++],b);}

int main(int c,char**v){
