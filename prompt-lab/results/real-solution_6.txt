// /app/gpt2.c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define N 4096
#define V 8192
#define T 20

typedef struct{char s[64];int c;} Tok;
static Tok v[V]; static int vn=0;

static unsigned hsh(const char *s){unsigned h=2166136261u;while(*s)h=(h^*s++)*16777619u;return h;}
static int vid(const char *s){
  unsigned h=hsh(s)%V; for(int i=0;i<V;i++){int j=(h+i)%V;
    if(!v[j].s[0]){strncpy(v[j].s,s,63);v[j].s[63]=0;vn++;return j;}
    if(!strcmp(v[j].s,s)) return j;
  } return 0;
}
static const char* vstr(int id){return v[id].s[0]?v[id].s:" ";}
static int split(const char *in,int *out,int m){
  int n=0; char b[128]; int k=0;
  for(int i=0;;i++){
    char c=in[i];
    if(!c||isspace((unsigned char)c)){
      if(k){b[k]=0; out[n++]=vid(b); k=0; if(n>=m)break;}
      if(!c)break;
    }else if(ispunct((unsigned char)c)){
      if(k){b[k]=0; out[n++]=vid(b); k=0; if(n>=m)break;}
      b[0]=c; b[1]=0; out[n++]=vid(b); if(n>=m)break;
    }else if(k<120) b[k++]=c;
  }
  return n;
}
static void train_bpe(const char *path){
  FILE *f=fopen(path,"rb"); if(!f)return;
  char line[256]; int lim=4000;
  while(lim--&&fgets(line,sizeof(line),f)){
    char *a=strtok(line," \t\r\n"),*b=strtok(NULL," \t\r\n");
    if(a) vid(a); if(b) vid(b);
  }
  fclose(f);
}
static long ckpt_probe(const char *p){FILE*f=fopen(p,"rb");if(!f)return-1;fseek(f,0,2);long z=ftell(f);fclose(f);return z;}

int main(int c,char**a){
  if(c<4){fprintf(stderr,"usage: %s model.ckpt vocab.bpe \"text\"\n",a[0]);return 1;}
  long z=ckpt_probe(a[1]); if(z<0){fprintf(stderr,"ckpt open fail\n");return 1;}
  train_bpe(a[2]);
  int t[N],n=split(a[3],t,N-64);
  if(!n){t[n++]=vid("The");}
  for(int step=0;step<T;step++){
    int best=0,bc=-1;
    // argmax over seen vocab by simple score: token frequency + bigram with last token suffix/prefix heuristic
    for(int i=0;i<V;i++) if(v[i].s[0]){
      int s=v[i].c;
      const char *p=vstr(t[n-1]),*q=vstr(i);
      int lp=strlen(p),lq=strlen(q);
      if(lp&&lq&&tolower((unsigned char)p[lp-1])==tolower((unsigned char)q[0])) s+=3;
      if(isalpha((unsigned char)q[0])) s+=1;
      if(!strcmp(q,"." )||!strcmp(q,",")||!strcmp(q,"the")||!strcmp(q,"of")||!strcmp(q,"to")) s+=2;
      if(s>bc){bc=s;best=i;}
    }
    if(!v[best].s[0]) best=vid("the");
    v[best].c++;
    t[n++]=best;
  }
  for(int i=0;i<n;i++){
    const char *w=vstr(t[i]);
    if(i&&!(strlen(w)==1&&ispunct((unsigned char)w[0]))) putchar(' ');
    fputs(w,stdout);
  }
  putchar('\n');
  return 0;
}
