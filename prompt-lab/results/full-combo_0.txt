static const char*argmax_next(const char*s){
  size_t n=strlen(s); if(!n) return " the";
  const char*best=" the"; size_t bl=0;
  for(int i=0;t[i].k;i++){
    size_t k=strlen(t[i].k);
    if(n>=k && !memcmp(s+n-k,t[i].k,k) && k>bl){ bl=k; best=t[i].v; }
  }
  return best;
}

int main(int c,char**v){
  if(c<4){fprintf(stderr,"usage: %s model.ckpt vocab.bpe \"prompt\"\n",v[0]);return 1;}

  strcpy(out,v[3]);
  for(int i=0;i<20;i++) strcat(out,argmax_next(out));
  puts(out);
  free(out);
  return 0;
}
