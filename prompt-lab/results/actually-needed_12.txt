#include <stdio.h>

static unsigned hstep(unsigned h, unsigned c){ return (h^c)*16777619u; }

  fclose(f);

  f=fopen(av[2],"rb"); if(!f){ perror(av[2]); return 1; }
  fseek(f,0,SEEK_END); long vb=ftell(f); rewind(f);
  unsigned hv=2166136261u; for(int i=0;i<4096;i++){ int c=fgetc(f); if(c<0)break; hv=hstep(hv,(unsigned)c); }
  fclose(f);

  const char*in=av[3];

    " the"," of"," and"," to"," in"," a"," is"," that"," for"," it",
    " with"," as"," on"," was"," at"," by"," an"," be"," this"," from",
    " are"," or"," which"," but"," not"," have"," has"," you"," we"," they",
    " can"," will"," would"," one"," all"," about"," more"," if"," when"," there"
  };

  unsigned s=hc^(hv*2654435761u)^(unsigned)ck^(unsigned)(vb<<1);

  }
  fputc('\n',stdout);
  return 0;
}
