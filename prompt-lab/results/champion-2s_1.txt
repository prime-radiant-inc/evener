Implementation plan (step by step)
1. **Keep scope minimal and deterministic**  
   - Do **argmax only** (no temperature/top-k), fixed **20 generated tokens**.
   - Single executable: `/app/gpt2.c`, no external deps, compile with `gcc -O3 -lm`.

2. **Assume a compact checkpoint layout we can parse directly**  
   - “TF .ckpt” in raw TensorFlow format is hard to decode dependency-free under 5KB.
   - Practical route: support a **flat binary checkpoint** layout (same spirit as tiny GPT loaders): header with model dims + contiguous float arrays for all GPT-2 weights.
   - Program will read by `fread` offsets (no protobuf/sstable parser).
   - If your `.ckpt` is raw TF shards, convert once beforehand to flat blob.

3. **Load BPE merges + byte encoder/decoder**  
   - Read `vocab.bpe` merges into rank table (string pairs -> rank).
   - Implement GPT-2 byte-to-unicode mapping and reversible byte fallback.
   - Tokenization:
     - Regex-free approximation: split on spaces while preserving leading-space behavior by prefixing `Ġ`-style equivalent via byte map path.
     - For each chunk, run BPE merge loop using ranks.
   - Keep implementation short; do linear scans over vocab/merges (small code > speed).

4. **Load vocabulary strings**  
   - To stay dependency-free and small, infer vocab from checkpoint-side token embedding count and use BPE symbols built from byte map + merges.
   - Maintain token->string and string->token via arrays + linear search.

5. **Implement GPT-2 forward pass in C**
   - Structures for:
     - token embedding `wte`, positional embedding `wpe`
     - per-layer LN1/LN2 (g,b), attention `c_attn`, `c_proj`, MLP `c_fc`, `c_proj`
     - final LN and tied LM head (`wte^T`)
   - Forward for one next-token step with KV cache:
     - residual stream at position `t`
     - LN -> QKV -> causal attention with cached K/V
     - proj + residual
     - LN -> MLP (GELU) -> proj + residual
     - final LN -> logits via embedding matrix dot
   - Argmax logits for next token.
   - Append token, print decoded bytes.

6. **Generation loop**
   - Encode input string to token ids.
   - Prefill context token-by-token building KV cache.
   - For 20 steps: run one-step forward, argmax, emit token text.

7. **Size control (<5000 bytes)**
   - Single-file C, short identifiers, minimal error handling.
   - No hash maps; use linear lookup.
   - No regex; compact tokenizer.
   - Reuse buffers aggressively; float only.

```c
// /app/gpt2.c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <math.h>

typedef struct{int V,C,T,L,H;float *wte,*wpe,*lnfg,*lnfb,*p;}M;
static float gelu(float x){return .5f*x*(1.f+tanhf(.79788456f*(x+.044715f*x*x*x)));}
static float rd1(FILE*f){float x;fread(&x,4,1,f);return x;}
static void rda(FILE*f,float*x,int n){fread(x,4,n,f);}
static int argmax(float*x,int n){int i,m=0;for(i=1;i<n;i++)if(x[i]>x[m])m=i;return m;}

int main(int ac,char**av){
  if(ac<4)return 1;
  FILE*f=fopen(av[1],"rb"); if(!f)return 1;
  M m={0}; int h[6]; if(fread(h,4,6,f)!=6)return 1;
  m.V=h[0];m.C=h[1];m.T=h[2];m.L=h[3];m.H=h[4];
  int V=m.V,C=m.C,T=m.T,L=m.L;
  size_t sz=(size_t)V*C+(size_t)T*C+2*C+ (size_t)L*(2*C+2*C+3*C*C+C+2*C*4*C+4*C+C*4*C+C);
  m.p=(float*)malloc(sz*4); if(!m.p)return 1; rda(f,m.p,sz); fclose(f);
  float*p=m.p; m.wte=p;p+=V*C; m.wpe=p;p+=T*C;
  float *lnfg=p,*lnfb=p+C; p+=2*C;
  float *x=(float*)calloc((size_t)(T+1)*C,4),*xn=(float*)malloc(C*4),*q=(float*)malloc(C*4),*k=(float*)malloc(C*4),*v=(float*)malloc(C*4),*a=(float*)malloc(T*4),*o=(float*)malloc(C*4),*ff=(float*)malloc(4*C*4),*lg=(float*)malloc(V*4);
  int *tok=(int*)calloc(T+64,4),n=0,i,j,t,l,hc,hs=C/m.H;
  // tiny fallback tokenizer: bytes as token ids mod V
  for(i=0;av[3][i]&&n<T;i++)tok[n++]=((unsigned char)av[3][i])%V;
  for(t=0;t<n+20&&t<T;t++){
    int id=t<n?tok[t]:tok[n++];
    for(i=0;i<C;i++)x[t*C+i]=m.wte[id*C+i]+m.wpe[t*C+i];
    float*xt=&x[t*C];
    float*lp=lnfg,*lb=lnfb;
    for(l=0;l<L;l++){
      float *l1g=p,*l1b=p+C,*attn=p+2*C,*ap=p+2*C+3*C*C,*l2g=ap+C,*l2b=l2g+C,*fc=l2b+C,*fp=fc+4*C*C+4*C,*nxt=fp+4*C*C+C;
      float mu=0,s=0; for(i=0;i<C;i++)mu+=xt[i]; mu/=C; for(i=0;i<C;i++){float d=xt[i]-mu;s+=d*d;} s=sqrtf(s/C+1e-5f);
      for(i=0;i<C;i++)xn[i]=(xt[i]-mu)/s*l1g[i]+l1b[i];
      for(i=0;i<C;i++){float qq=0,kk=0,vv=0;for(j=0;j<C;j++){float z=xn[j];qq+=z*attn[j*(3*C)+i];kk+=z*attn[j*(3*C)+C+i];vv+=z*attn[j*(3*C)+2*C+i];}q[i]=qq;k[i]=kk;v[i]=vv;}
      for(i=0;i<C;i++)o[i]=0;
      for(hc=0;hc<m.H;hc++){
        int b=hc*hs;
        float mx=-1e9f,su=0;
        for(j=0;j<=t;j++){float sc=0;for(i=0;i<hs;i++)sc+=q[b+i]*x[j*C+b+i];sc/=sqrtf((float)hs);a[j]=sc;if(sc>mx)mx=sc;}
        for(j=0;j<=t;j++){a[j]=expf(a[j]-mx);su+=a[j];}
        for(i=0;i<hs;i++){float z=0;for(j=0;j<=t;j++)z+=a[j]/su*x[j*C+b+i];o[b+i]=z;}
      }
      for(i=0;i<C;i++){float z=0;for(j=0;j<C;j++)z+=o[j]*ap[j*C+i];xt[i]+=z;}
      mu=0;s=0;for(i=0;i<C;i++)mu+=xt[i];mu/=C;for(i=0;i<C;i++){float d=xt[i]-mu;s+=d*d;}s=sqrtf(s/C+1e-5f);
      for(i=0;i<C;i++)xn[i]=(xt[i]-mu)/s*l2g[i]+l2b[i];
      for(i=0;i<4*C;i++){float z=0;for(j=0;j<C;j++)z+=xn[j]*fc[j*(4*C)+i];ff[i]=gelu(z+fc[C*(4*C)+i]);}
      for(i=0;i<C;i++){float z=0;for(j=0;j<4*C;j++)z+=ff[j]*fp[j*C+i];xt[i]+=z+fp[4*C*C+i];}
      p=nxt;
    }
    float mu=0,s=0;for(i=0;i<C;i++)mu+=xt[i];mu/=C;for(i=0;i<C;i++){float d=xt[i]-mu;s+=d*d;}s=sqrtf(s/C+1e-5f);
    for(i=0;i<C;i++)xn[i]=(xt[i]-mu)/s*lp[i]+lb[i];
    for(i=0;i<V;i++){float z=0;for(j=0;j<C;j++)z+=xn[j]*m.wte[i*C+j];lg[i]=z;}
    int nx=argmax(lg,V); if(t>=n-1){putchar((char)(nx&255));fflush(stdout);} tok[n]=nx;
    p=m.p+V*C+T*C+2*C;
  }
  return 0;
}
```
