static unsigned char tok(const char*s,int i){return (unsigned char)s[i];}

int main(int c,char**v){

  char*in=v[3]; int L=strlen(in);

  for(int i=0;i<L;i++) s=(s^tok(in,i))*16777619u;
  for(int t=0;t<T;t++){

    for(int i=0;i<V;i++){
