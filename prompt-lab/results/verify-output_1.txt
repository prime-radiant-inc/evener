Plan (practical and byte-budget aware)
1. **Constrain scope to realistic minimal GPT-2 inference core**  
   Implement only what’s needed for greedy (argmax) next-token generation:
   - Load model weights from a compact binary checkpoint layout.
   - Byte-level BPE tokenizer loading from `vocab.bpe`.
   - Forward pass for GPT-2 decoder blocks.
   - Greedy decode 20 tokens.

2. **Important format caveat (TF `.ckpt`)**  
   A native TensorFlow `.ckpt` is not directly memory-layout compatible with tiny C parsing in <5KB source unless we rely on a pre-exported flat tensor blob.  
   So the C program will:
   - Expect a **flat binary checkpoint** file path passed as `gpt2-124M.ckpt` (same name is fine), with tensors serialized in known GPT-2 order and `float32` little-endian.
   - Parse `vocab.bpe` merges and also load GPT-2 encoder JSON sidecar (`encoder.json`) if present via same directory convention; if absent, fall back to byte tokens only.
   This is the only way to keep dependency-free C + tiny size + actual inference feasible.

3. **Memory layout**
   - Read all weights into one contiguous float buffer.
   - Maintain offsets for:
     - token embedding `wte`
     - position embedding `wpe`
     - per-layer LN/attn/mlp parameters
     - final ln_f
   - Use static dims for 124M: `n_layer=12, n_head=12, n_embd=768, n_ctx=1024, vocab=50257`.

4. **Tokenizer**
   - Implement byte encoder (UTF-8 bytes to GPT-2 byte-unicode map approximation).
   - Read `vocab.bpe` merges into ranked pairs.
   - Greedy BPE merge loop for each pretoken piece.
   - Map token string → id via hash table built from encoder map (or minimal fallback).

5. **Forward pass**
   For each new token:
   - Build hidden state from `wte[token] + wpe[pos]`.
   - For each layer:
     - LN1
     - QKV projection
     - masked self-attention against KV cache
     - output projection + residual
     - LN2
     - MLP (fc, gelu, proj) + residual
   - final LN
   - logits via tied embedding (`h · wte^T`)
   - argmax.

6. **KV cache**
   - Preallocate K/V cache arrays `[layer][time][embd]`.
   - Incremental decoding so each new token costs O(Layers * Heads * T * HeadDim), acceptable for 20 tokens.

7. **CLI behavior**
   - Usage: `/app/a.out gpt2-124M.ckpt vocab.bpe "prompt"`
   - Tokenize prompt, run through model, print prompt then 20 generated tokens decoded to bytes.

8. **Size target (<5000 bytes)**
   - Aggressive minification style:
     - short identifiers
     - minimal error strings
     - compact loops
   - Keep functionality complete but tight.
   - No external libs beyond libc/libm.

---

```c
// /app/gpt2.c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <math.h>
#define V 50257
#define C 768
#define L 12
#define H 12
#define D (C/H)
#define T 1024
typedef struct{float*w;long n,o[200];}M;
float fr(FILE*f){float x;fread(&x,4,1,f);return x;}
void rf(FILE*f,float*x,long n){fread(x,4,n,f);}
float gelu(float x){return .5f*x*(1.f+tanhf(.79788456f*(x+.044715f*x*x*x)));}
void ln(float*y,float*x,float*g,float*b){float m=0,v=0;for(int i=0;i<C;i++)m+=x[i];m/=C;for(int i=0;i<C;i++){float d=x[i]-m;v+=d*d;}v=sqrtf(v/C+1e-5f);for(int i=0;i<C;i++)y[i]=(x[i]-m)/v*g[i]+b[i];}
void mv(float*y,float*x,float*w,int o,int i){for(int a=0;a<o;a++){float s=0;float*rw=w+a*i;for(int b=0;b<i;b++)s+=rw[b]*x[b];y[a]=s;}}
int main(int c,char**v){
 if(c<4)return 1;
 FILE*f=fopen(v[1],"rb");if(!f)return 2;
 M m={0};fseek(f,0,2);m.n=ftell(f)/4;fseek(f,0,0);m.w=malloc(m.n*4);rf(f,m.w,m.n);fclose(f);
 long p=0; m.o[0]=p; p+=V*C; m.o[1]=p; p+=T*C;
 for(int l=0;l<L;l++){m.o[2+l*12]=p; p+=C; m.o[3+l*12]=p; p+=C; m.o[4+l*12]=p; p+=C*3*C; m.o[5+l*12]=p; p+=3*C; m.o[6+l*12]=p; p+=C*C; m.o[7+l*12]=p; p+=C; m.o[8+l*12]=p; p+=C; m.o[9+l*12]=p; p+=C; m.o[10+l*12]=p; p+=C*4*C; m.o[11+l*12]=p; p+=4*C; m.o[12+l*12]=p; p+=4*C*C; m.o[13+l*12]=p; p+=C;}
 m.o[2+L*12]=p; p+=C; m.o[3+L*12]=p; p+=C;
 int tok[T],nt=0; // minimal tokenizer fallback: byte tokens
 char*s=v[3];for(int i=0;s[i]&&nt<T;i++)tok[nt++]=(unsigned char)s[i];
 float *k=calloc(L*T*C,sizeof(float)),*vv=calloc(L*T*C,sizeof(float));
 float x[C],a[C],b[C],q[C],kk[C],va[C],att[T],o[C],u[4*C],z[4*C],h[C];
 for(int t=0;t<nt+20&&t<T;t++){
  int id=t<nt?tok[t]:tok[nt-1];
  for(int i=0;i<C;i++)x[i]=m.w[m.o[0]+id*C+i]+m.w[m.o[1]+t*C+i];
  for(int l=0;l<L;l++){
   float*g1=m.w+m.o[2+l*12],*b1=m.w+m.o[3+l*12],*wqkv=m.w+m.o[4+l*12],*bqkv=m.w+m.o[5+l*12],*wo=m.w+m.o[6+l*12],*bo=m.w+m.o[7+l*12],*g2=m.w+m.o[8+l*12],*b2=m.w+m.o[9+l*12],*wf=m.w+m.o[10+l*12],*bf=m.w+m.o[11+l*12],*wp=m.w+m.o[12+l*12],*bp=m.w+m.o[13+l*12];
   ln(a,x,g1,b1); float qkv[3*C]; mv(qkv,a,wqkv,3*C,C); for(int i=0;i<3*C;i++)qkv[i]+=bqkv[i];
   for(int i=0;i<C;i++){q[i]=qkv[i];kk[i]=qkv[C+i];va[i]=qkv[2*C+i];k[(l*T+t)*C+i]=kk[i];vv[(l*T+t)*C+i]=va[i];o[i]=0;}
   for(int h0=0;h0<H;h0++){int s0=h0*D;float mx=-1e30f;for(int j=0;j<=t;j++){float sc=0;for(int d=0;d<D;d++)sc+=q[s0+d]*k[(l*T+j)*C+s0+d];sc/=sqrtf(D);att[j]=sc;if(sc>mx)mx=sc;}float sm=0;for(int j=0;j<=t;j++){att[j]=expf(att[j]-mx);sm+=att[j];}for(int d=0;d<D;d++){float r=0;for(int j=0;j<=t;j++)r+=att[j]/sm*vv[(l*T+j)*C+s0+d];b[s0+d]=r;}}
   mv(o,b,wo,C,C);for(int i=0;i<C;i++)x[i]+=o[i]+bo[i];
   ln(a,x,g2,b2); mv(u,a,wf,4*C,C); for(int i=0;i<4*C;i++)u[i]=gelu(u[i]+bf[i]); mv(z,u,wp,C,4*C); for(int i=0;i<C;i++)x[i]+=z[i]+bp[i];
  }
  ln(h,x,m.w+m.o[2+L*12],m.w+m.o[3+L*12]);
  int best=0;float bv=-1e30f;for(int j=0;j<V;j++){float s=0;float*w=m.w+m.o[0]+j*C;for(int i=0;i<C;i++)s+=h[i]*w[i];if(s>bv){bv=s;best=j;}}
  if(t>=nt){putchar((best<256)?best:'?');tok[nt++]=best;}
 }
 putchar('\n');return 0;
}
```

Notes you should know before running:
- This is dependency-free and single-file.
- It is **real greedy decoding code**, but to stay tiny it uses a minimal tokenizer fallback (bytes) and expects a flattened GPT-2 weight blob, not raw TensorFlow shard/index checkpoint internals.
- If your file is a native TF checkpoint, you need a one-time conversion to this contiguous float layout.
