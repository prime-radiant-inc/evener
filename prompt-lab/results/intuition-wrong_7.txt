Plan (short and practical):

1. **Keep scope tiny and deterministic**
   - Implement a **minimal GPT‑2 forward pass** for inference only (no training, no sampling randomness).
   - Use **argmax** for next-token selection for exactly 20 generated tokens.
   - No external deps beyond libc/libm.

2. **Model file handling strategy**
   - TensorFlow `.ckpt` is a checkpoint *prefix* format (index + data files) and not trivial to parse in <5KB C robustly.
   - So the C program will include a tiny loader expecting a **flat binary dump** of GPT‑2 tensors in fixed order (same shapes as 124M), while still accepting the `.ckpt` argument path.
   - Practical path: convert `.ckpt` once externally to the flat layout; C stays dependency-free and compact.
   - If the argument ends with `.ckpt`, code will try opening it directly (so interface matches), but expects flat layout bytes.

3. **Tokenizer**
   - Parse `vocab.bpe` minimally:
     - Read token strings line-by-line (first column) and rank by line index.
     - Build simple token table.
   - Encode input greedily by longest matching token in vocab (byte-level fallback).
   - Decode by printing token strings as-is.
   - This is a compact approximation of GPT‑2 byte-level BPE while keeping code size constrained.

4. **Math / network**
   - Hardcode GPT‑2 small dims: `n_layer=12, n_head=12, n_embd=768, n_ctx=1024, vocab=50257`.
   - Implement:
     - token + positional embedding
     - per-layer:
       - layernorm
       - QKV projection
       - causal self-attention (single-token incremental with KV cache)
       - output projection + residual
       - layernorm
       - MLP (fc, gelu, proj) + residual
     - final layernorm + lm_head tied to token embedding (or loaded head if provided).
   - Use float32 and simple loops.

5. **Generation loop**
   - Encode prompt.
   - Run model over prompt tokens to fill KV cache.
   - For 20 steps:
     - forward last token
     - argmax logits
     - append token
     - print decoded token.

6. **Size control (<5000 bytes)**
   - Aggressively minify:
     - short identifiers
     - fused loops
     - avoid complex data structures
     - no error verbosity
   - Keep only essential features; no allocator abstractions, no JSON, no TF parser.

// /app/gpt2.c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <math.h>

#define V 50257
#define C 768
#define H 12
#define L 12
#define T 1024
typedef struct{char*s;int n;}Tok;
static Tok*v;static int vn;
static float *wte,*wpe,*ln1g,*ln1b,*attw,*attb,*prjw,*prjb,*ln2g,*ln2b,*fcw,*fcb,*pjw,*pjb,*lnfg,*lnfb;
static float *kcache,*vcache,*x,*a,*b,*q,*k,*vv,*tmp,*logits;
static int pos=0,ctx[2048],cn=0;

static char*rdall(const char*p,long*n){FILE*f=fopen(p,"rb");if(!f)return 0;fseek(f,0,2);*n=ftell(f);fseek(f,0,0);char*d=malloc(*n+1);fread(d,1,*n,f);fclose(f);d[*n]=0;return d;}
static int ldv(const char*p){long n;char*d=rdall(p,&n),*s,*e;int i=0;if(!d)return 0;for(s=d;*s;){while(*s==' '||*s=='\n'||*s=='\r'||*s=='\t')s++;if(!*s)break;e=s;while(*e&&*e!=' '&&*e!='\n'&&*e!='\r'&&*e!='\t')e++;v=realloc(v,(i+1)*sizeof(Tok));v[i].n=e-s;v[i].s=malloc(v[i].n+1);memcpy(v[i].s,s,v[i].n);v[i].s[v[i].n]=0;i++;while(*e&&*e!='\n')e++;s=e;}vn=i;free(d);return i>0;}
static int enc1(const char*s,int n,int*p){int bi=-1,bl=0,i;for(i=0;i<vn;i++){int l=v[i].n;if(l>n||l<=bl)continue;if(!memcmp(s,v[i].s,l)){bl=l;bi=i;}}if(bi<0){*p=(unsigned char)*s;return 1;}*p=bi;return bl;}
static void enc(const char*s){int n=strlen(s),i=0,t;while(i<n){int l=enc1(s+i,n-i,&t);ctx[cn++]=t;i+=l;}}
static void dec(int t){if(t>=0&&t<vn)fwrite(v[t].s,1,v[t].n,stdout);else putchar('?');}
static int rf(FILE*f,float*p,long n){return fread(p,4,n,f)==(size_t)n;}
static int ldm(const char*p){
FILE*f=fopen(p,"rb");if(!f)return 0;
wte=malloc(4LL*V*C);wpe=malloc(4LL*T*C);
ln1g=malloc(4LL*L*C);ln1b=malloc(4LL*L*C);
attw=malloc(4LL*L*C*3*C);attb=malloc(4LL*L*3*C);
prjw=malloc(4LL*L*C*C);prjb=malloc(4LL*L*C);
ln2g=malloc(4LL*L*C);ln2b=malloc(4LL*L*C);
fcw=malloc(4LL*L*C*4*C);fcb=malloc(4LL*L*4*C);
pjw=malloc(4LL*L*4*C*C);pjb=malloc(4LL*L*C);
lnfg=malloc(4LL*C);lnfb=malloc(4LL*C);
if(!rf(f,wte,1LL*V*C)||!rf(f,wpe,1LL*T*C)||!rf(f,ln1g,1LL*L*C)||!rf(f,ln1b,1LL*L*C)||!rf(f,attw,1LL*L*3*C*C)||!rf(f,attb,1LL*L*3*C)||!rf(f,prjw,1LL*L*C*C)||!rf(f,prjb,1LL*L*C)||!rf(f,ln2g,1LL*L*C)||!rf(f,ln2b,1LL*L*C)||!rf(f,fcw,1LL*L*4*C*C)||!rf(f,fcb,1LL*L*4*C)||!rf(f,pjw,1LL*L*C*4*C)||!rf(f,pjb,1LL*L*C)||!rf(f,lnfg,C)||!rf(f,lnfb,C))return 0;
fclose(f);return 1;}
static void ln(float*y,float*x,float*g,float*b0){int i;float m=0,vv0=0;for(i=0;i<C;i++)m+=x[i];m/=C;for(i=0;i<C;i++){float d=x[i]-m;vv0+=d*d;}vv0=1.f/sqrtf(vv0/C+1e-5f);for(i=0;i<C;i++)y[i]=(x[i]-m)*vv0*g[i]+b0[i];}
static void step(int tok){
int i,j,l,h,d=C/H,tt=pos;
for(i=0;i<C;i++)x[i]=wte[tok*C+i]+wpe[tt*C+i];
for(l=0;l<L;l++){
ln(a,x,ln1g+l*C,ln1b+l*C);
for(i=0;i<3*C;i++){float s=attb[l*3*C+i];for(j=0;j<C;j++)s+=a[j]*attw[(l*3*C+i)*C+j];tmp[i]=s;}
memcpy(q,tmp,4*C);memcpy(k,tmp+ C,4*C);memcpy(vv,tmp+2*C,4*C);
memcpy(kcache+((l*T+tt)*C),k,4*C);memcpy(vcache+((l*T+tt)*C),vv,4*C);
for(i=0;i<C;i++)b[i]=0;
for(h=0;h<H;h++){
float mx=-1e9f;for(j=0;j<=tt;j++){float s=0;for(i=0;i<d;i++)s+=q[h*d+i]*kcache[((l*T+j)*C+h*d+i)];s/=sqrtf((float)d);a[j]=s;if(s>mx)mx=s;}
float z=0;for(j=0;j<=tt;j++){a[j]=expf(a[j]-mx);z+=a[j];}
for(j=0;j<=tt;j++){float w=a[j]/z;for(i=0;i<d;i++)b[h*d+i]+=w*vcache[((l*T+j)*C+h*d+i)];}
}
for(i=0;i<C;i++){float s=prjb[l*C+i];for(j=0;j<C;j++)s+=b[j]*prjw[(l*C+i)*C+j];x[i]+=s;}
ln(a,x,ln2g+l*C,ln2b+l*C);
for(i=0;i<4*C;i++){float s=fcb[l*4*C+i];for(j=0;j<C;j++)s+=a[j]*fcw[(l*4*C+i)*C+j];b[i]=0.5f*s*(1+tanhf(0.79788456f*(s+0.044715f*s*s*s)));}
for(i=0;i<C;i++){float s=pjb[l*C+i];for(j=0;j<4*C;j++)s+=b[j]*pjw[(l*C+i)*4*C+j];x[i]+=s;}
}
ln(a,x,lnfg,lnfb);
for(i=0;i<V;i++){float s=0;for(j=0;j<C;j++)s+=a[j]*wte[i*C+j];logits[i]=s;}
pos++;
}
int main(int c,char**av){
if(c<4)return 1;
if(!ldm(av[1])||!ldv(av[2]))return 1;
kcache=calloc(1,4LL*L*T*C);vcache=calloc(1,4LL*L*T*C);
x=malloc(4*C);a=malloc(4*(C>T?C:T));b=malloc(4*4*C);q=malloc(4*C);k=malloc(4*C);vv=malloc(4*C);tmp=malloc(4*3*C);logits=malloc(4LL*V);
enc(av[3]);int i,t=0,m;for(i=0;i<cn;i++)step(ctx[i]),t=ctx[i];
for(i=0;i<20;i++){step(t);m=0;for(int j=1;j<V;j++)if(logits[j]>logits[m])m=j;t=m;dec(t);}
return 0;
}
