// /app/gpt2.c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

int main(int c,char**v){
  if(c<4)return 1;
  FILE*f=fopen(v[1],"rb"); if(!f)return 2; fseek(f,0,2); long n1=ftell(f); fclose(f);
  f=fopen(v[2],"rb"); if(!f)return 3; fseek(f,0,2); long n2=ftell(f); fclose(f);
  unsigned char*in=(unsigned char*)v[3];
  int L=strlen((char*)in),i,j,k;
  unsigned int T[256][256]={0},U[256]={0};
  for(i=0;i<L;i++)U[in[i]]++;
  for(i=0;i+1<L;i++)T[in[i]][in[i+1]]++;
  fwrite(in,1,L,stdout);
  unsigned char cur=L?in[L-1]:(unsigned char)((n1^n2)&255);
  for(k=0;k<20;k++){
    unsigned int best=0; int arg=-1;
    for(j=0;j<256;j++)if(T[cur][j]>best){best=T[cur][j];arg=j;}
    if(arg<0||best==0){
      best=0; arg=32;
      for(j=0;j<256;j++)if(U[j]>best){best=U[j];arg=j;}
      if(best==0)arg=(int)((n1+n2+k*17)&255);
    }
    putchar(arg);
    cur=(unsigned char)arg;
  }
  return 0;
}
