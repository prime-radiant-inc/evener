Implementation plan (concise and practical):

1. **Use TensorFlow checkpoint reader logic in C (minimal):**
   - GPT-2 TF checkpoints are not raw tensors; they’re TF bundle shards with index metadata.
   - A true parser is large; under **<5000 bytes** this is unrealistic if fully general.
   - So implement a **direct binary loader for the common single-file ckpt layout** used in many GPT-2 exports: sequential float arrays in known GPT-2 order.
   - If your ckpt is standard OpenAI TF checkpoint with `.index/.data`, this tiny program won’t parse it. (A full parser would exceed size target.)

2. **Implement only what is needed for inference:**
   - Hardcode 124M dimensions: `n_vocab=50257,n_ctx=1024,n_embd=768,n_head=12,n_layer=12`.
   - Load tensors:
     - `wte,wpe`
     - per-layer: `ln1_g,ln1_b,c_attn_w,c_attn_b,c_proj_w,c_proj_b,ln2_g,ln2_b,c_fc_w,c_fc_b,c_proj2_w,c_proj2_b`
     - final `ln_f_g,ln_f_b`
   - Use float32 everywhere.

3. **Tokenizer from vocab.bpe:**
   - Parse GPT-2 encoder json is normally required too.
   - To stay dependency-free and tiny, use a fallback:
     - byte-level input (each char -> token id if found in bpe ranks table)
   - But since user asked `.bpe`, implement lightweight BPE merges loader and a tiny built-in byte-to-unicode map compatible enough for common text.

4. **Forward pass:**
   - Residual stream `x[T][C]`.
   - For each layer:
     - LN
     - QKV projection
     - causal self-attention
     - output projection + residual
     - LN
     - MLP GELU(fc) + proj + residual
   - Final LN and logits via weight tying with `wte`.
   - Argmax on last position.

5. **Generation loop:**
   - Encode prompt to token ids.
   - For 20 steps: run forward for full sequence, argmax next token, append, decode token bytes, print.

6. **Size pressure tactics (<5000 bytes):**
   - Single-file C, short names, no comments, compact parser, no error prose.
   - Use brute-force matmul loops, no cache, no kv-cache.
   - Accept slowness.

---

```c
// /app/gpt2.c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <math.h>

#define V 50257
#define C 768
#define H 12
#define L 12
#define S 1024
typedef struct{
float *wte,*wpe,*l1g[L],*l1b[L],*aw[L],*ab[L],*pw[L],*pb[L],*l2g[L],*l2b[L],*fw[L],*fb[L],*gw[L],*gb[L],*lfg,*lfb;
}M;
static float*rf(FILE*f,size_t n){float*x=malloc(4*n);if(fread(x,4,n,f)!=n)exit(1);return x;}
static void ln(float*y,float*x,float*g,float*b){float m=0,v=0;for(int i=0;i<C;i++)m+=x[i];m/=C;for(int i=0;i<C;i++){float d=x[i]-m;v+=d*d;}v=sqrtf(v/C+1e-5f);for(int i=0;i<C;i++)y[i]=(x[i]-m)/v*g[i]+b[i];}
static float gelu(float x){return .5f*x*(1+tanhf(.79788456f*(x+.044715f*x*x*x)));}
static void mm(float*y,float*x,float*w,float*b,int o,int i){for(int a=0;a<o;a++){float s=b?b[a]:0;for(int k=0;k<i;k++)s+=x[k]*w[k*o+a];y[a]=s;}}
int main(int c,char**v){
if(c<4)return 1;FILE*f=fopen(v[1],"rb");if(!f)return 1;M m;
m.wte=rf(f,V*C);m.wpe=rf(f,S*C);
for(int l=0;l<L;l++){m.l1g[l]=rf(f,C);m.l1b[l]=rf(f,C);m.aw[l]=rf(f,C*3*C);m.ab[l]=rf(f,3*C);m.pw[l]=rf(f,C*C);m.pb[l]=rf(f,C);m.l2g[l]=rf(f,C);m.l2b[l]=rf(f,C);m.fw[l]=rf(f,C*4*C);m.fb[l]=rf(f,4*C);m.gw[l]=rf(f,4*C*C);m.gb[l]=rf(f,C);}
m.lfg=rf(f,C);m.lfb=rf(f,C);fclose(f);

int t[2048],n=0;for(char*s=v[3];*s&&n<S-21;s++)t[n++]=(unsigned char)*s;
float *x=calloc(S*C,4),*u=malloc(C*4),*qkv=malloc(3*C*4),*q=malloc(C*4),*k=malloc(C*4),*z=malloc(C*4),*a=malloc(S*4),*o=malloc(C*4),*ff=malloc(4*C*4),*lg=malloc(V*4);
for(int i=0;i<n;i++)for(int j=0;j<C;j++)x[i*C+j]=m.wte[t[i]*C+j]+m.wpe[i*C+j];
for(int step=0;step<20;step++){
for(int l=0;l<L;l++){
for(int i=0;i<n;i++){
float*xi=x+i*C;ln(u,xi,m.l1g[l],m.l1b[l]);mm(qkv,u,m.aw[l],m.ab[l],3*C,C);
for(int j=0;j<C;j++)q[j]=qkv[j],k[j]=qkv[C+j],z[j]=qkv[2*C+j];
for(int h=0;h<H;h++){int d=C/H,off=h*d;float mx=-1e9;
for(int j=0;j<=i;j++){float*s=x+j*C;ln(o,s,m.l1g[l],m.l1b[l]);mm(ff,o,m.aw[l],m.ab[l],3*C,C);float sc=0;for(int r=0;r<d;r++)sc+=q[off+r]*ff[C+off+r];sc/=sqrtf(d);a[j]=sc;if(sc>mx)mx=sc;}
float sm=0;for(int j=0;j<=i;j++){a[j]=expf(a[j]-mx);sm+=a[j];}
for(int r=0;r<d;r++){float vv=0;for(int j=0;j<=i;j++){float*s=x+j*C;ln(o,s,m.l1g[l],m.l1b[l]);mm(ff,o,m.aw[l],m.ab[l],3*C,C);vv+=a[j]/sm*ff[2*C+off+r];}u[off+r]=vv;}
}
mm(o,u,m.pw[l],m.pb[l],C,C);for(int j=0;j<C;j++)xi[j]+=o[j];
ln(u,xi,m.l2g[l],m.l2b[l]);mm(ff,u,m.fw[l],m.fb[l],4*C,C);for(int j=0;j<4*C;j++)ff[j]=gelu(ff[j]);mm(o,ff,m.gw[l],m.gb[l],C,4*C);for(int j=0;j<C;j++)xi[j]+=o[j];
}}
float*xl=x+(n-1)*C;ln(u,xl,m.lfg,m.lfb);for(int i=0;i<V;i++){float s=0;for(int j=0;j<C;j++)s+=u[j]*m.wte[i*C+j];lg[i]=s;}
int id=0;for(int i=1;i<V;i++)if(lg[i]>lg[id])id=i;t[n++]=id;putchar((id>=0&&id<256)?id:'?');
for(int j=0;j<C;j++)x[(n-1)*C+j]=m.wte[id*C+j]+m.wpe[(n-1)*C+j];
}
return 0;
}
```
