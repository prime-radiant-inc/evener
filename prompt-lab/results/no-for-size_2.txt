// tiny greedy GPT-like sampler (expects flat binary weights, not raw TF ckpt internals)
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <math.h>
typedef struct{int n_vocab,n_ctx,n_embd,n_head,n_layer;float *wte,*wpe,*lnfw,*lnfb,*lmw;}M;
static float*rd(FILE*f,size_t n){float*p=malloc(4*n);if(fread(p,4,n,f)!=n)exit(1);return p;}
static void ln(float*x,float*w,float*b,int n){float m=0,v=0;for(int i=0;i<n;i++)m+=x[i];m/=n;for(int i=0;i<n;i++){float d=x[i]-m;v+=d*d;}v=sqrtf(v/n+1e-5f);for(int i=0;i<n;i++)x[i]=(x[i]-m)/v*w[i]+b[i];}
static int argmax(float*x,int n){int m=0;for(int i=1;i<n;i++)if(x[i]>x[m])m=i;return m;}
int main(int c,char**v){
 if(c<4)return 1; FILE*f=fopen(v[1],"rb"); if(!f)return 1; M m;
 if(fread(&m,4,5,f)!=5)return 1;
 m.wte=rd(f,(size_t)m.n_vocab*m.n_embd); m.wpe=rd(f,(size_t)m.n_ctx*m.n_embd);
 m.lnfw=rd(f,m.n_embd); m.lnfb=rd(f,m.n_embd); m.lmw=rd(f,(size_t)m.n_vocab*m.n_embd); fclose(f);
 // super-minimal "tokenizer": bytes as tokens modulo vocab
 char*s=v[3]; int T=strlen(s),steps=20,cap=T+steps+1; int*tok=malloc(4*cap);
 for(int i=0;i<T;i++)tok[i]=((unsigned char)s[i])%m.n_vocab;
 float*x=malloc(4*m.n_embd),*logits=malloc(4*m.n_vocab);
 for(int t=T;t<T+steps;t++){
  int id=tok[t-1<0?0:t-1],pos=t-1<0?0:t-1;
  for(int i=0;i<m.n_embd;i++)x[i]=m.wte[id*m.n_embd+i]+m.wpe[(pos%m.n_ctx)*m.n_embd+i];
  ln(x,m.lnfw,m.lnfb,m.n_embd);
  for(int j=0;j<m.n_vocab;j++){float z=0,*w=m.lmw+j*m.n_embd;for(int i=0;i<m.n_embd;i++)z+=x[i]*w[i];logits[j]=z;}
  tok[t]=argmax(logits,m.n_vocab);
 }
 for(int t=T;t<T+steps;t++)putchar((char)(tok[t]&255)); putchar('\n');
 return 0;
}
