static char*slurp(const char*p,long*n){FILE*f=fopen(p,"rb");if(!f)return 0;fseek(f,0,2);*n=ftell(f);fseek(f,0,0);char*b=malloc(*n+1);if(!b){fclose(f);return 0;}fread(b,1,*n,f);b[*n]=0;fclose(f);return b;}
static int split(char*s,char**a,int m){int n=0;for(;*s&&n<m;){while(*s=='\n'||*s=='\r')*s++=0;if(!*s)break;a[n++]=s;while(*s&&*s!='\n'&&*s!='\r')s++;}return n;}
