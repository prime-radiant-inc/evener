#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <math.h>

#define MAXV 60000
#define MAXTOK 4096
#define GEN 20

typedef struct{char*s;int id;}Tok;
static Tok tv[MAXV]; static int nv;
static char*strdup2(const char*s){size_t n=strlen(s)+1;char*r=malloc(n);memcpy(r,s,n);return r;}

static void load_bpe(const char*fn){
  FILE*f=fopen(fn,"rb"); if(!f){perror(fn);exit(1);}
  char line[1024]; int id=0;
  while(fgets(line,sizeof(line),f)&&id<MAXV){
    if(line[0]=='#'||line[0]=='\n')continue;
    char*a=strtok(line," \t\r\n"),*b=strtok(NULL," \t\r\n");
    if(!a||!b)continue;
    tv[id].s=strdup2(a); tv[id].id=id; id++;
  }
  nv=id; fclose(f);
}

static int tok_find(const char*s){
  for(int i=0;i<nv;i++) if(!strcmp(tv[i].s,s)) return i;
  return 0;
}
static int simple_tokenize(const char*in,int*out){
  int n=0; char*tmp=strdup2(in),*p=strtok(tmp," ");
  while(p&&n<MAXTOK){ out[n++]=tok_find(p); p=strtok(NULL," "); }
  free(tmp); return n;
}
static const char*tok_str(int id){ return (id>=0&&id<nv&&tv[id].s)?tv[id].s:""; }

int main(int c,char**v){
  if(c<4){fprintf(stderr,"usage: %s model.ckpt vocab.bpe \"text\"\n",v[0]);return 1;}

  for(int i=0;i<n;i++){ if(i)printf(" "); printf("%s",tok_str(ctx[i])); }

  // Placeholder argmax continuation: deterministic next-id = (last+1)%nv

  for(int t=0;t<GEN;t++){
    int nxt=(cur+1)% (nv?nv:1);
    printf("%s%s", n||t?" ":"", tok_str(nxt));
    cur=nxt;
  }
  printf("\n");
  return 0;
}
