```c
// /app/gpt2.c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <math.h>

#define T 20
#define V 50257
#define N 4096

static unsigned x=1;
static float fr(){x=1664525u*x+1013904223u;return (x>>8)*(1.0f/16777216.0f);}

int main(int c,char**v){
  if(c<4){fprintf(stderr,"usage: %s model.ckpt vocab.bpe \"prompt\"\n",v[0]);return 1;}
  FILE*fm=fopen(v[1],"rb"); if(!fm){perror("ckpt"); return 1;}
  FILE*fb=fopen(v[2],"rb"); if(!fb){perror("bpe"); return 1;}
  fseek(fm,0,SEEK_END); long ms=ftell(fm); fseek(fm,0,SEEK_SET);
  fseek(fb,0,SEEK_END); long bs=ftell(fb); fseek(fb,0,SEEK_SET);
  char *bb=malloc(bs+1); fread(bb,1,bs,fb); bb[bs]=0; fclose(fb);
  // minimal prompt echo
  char *p=v[3]; fputs(p,stdout);
  // placeholder token loop with argmax over synthetic logits
  for(int t=0;t<T;t++){
    int ai=0; float av=-1e9f;
    for(int i=0;i<V;i++){
      float z=fr()-0.5f; // TODO: real logits from transformer forward pass
      if(z>av){av=z;ai=i;}
    }
    // naive printable token stand-in
    char out[32]; snprintf(out,sizeof(out)," <%d>",ai);
    fputs(out,stdout);
  }
  fputc('\n',stdout);
  free(bb); fclose(fm);
  return 0;
}
```
