Implementation plan (brief and practical):

1. **Assume a simple `.ckpt` format is not directly parsable in tiny C**
   - Native TensorFlow `.ckpt` is a complex container (index + shards / tensor bundle, protobuf-like metadata), not a flat float dump.
   - A strict `<5000 bytes` single-file C program cannot realistically include a robust TF checkpoint parser plus GPT‑2 inference plus BPE.

2. **Use a compact “ckpt-as-flat-binary” path**
   - Implement a tiny GPT‑2 forward pass in C with no deps.
   - Read weights from a single binary stream in fixed GPT‑2 order (embeddings, blocks, ln_f, lm_head).
   - Keep code very small and argmax-only generation for 20 steps.

3. **Implement minimal BPE/tokenizer handling**
   - Parse `vocab.bpe` merges (rank list).
   - Use GPT‑2 byte-to-unicode trick in simplified form.
   - Greedy pair-merge for each tokenized chunk.
   - Keep a small hash-table-ish linear lookup (size/perf tradeoff for tiny code).

4. **Model architecture supported**
   - GPT‑2 small (124M): n_layer=12, n_head=12, n_embd=768, n_ctx<=1024.
   - Causal self-attention + MLP GeLU + residual + LayerNorm.
   - Argmax sampling for next token.
   - Continue exactly 20 new tokens.

5. **CLI and behavior**
   - Run: `/app/a.out gpt2-124M.ckpt vocab.bpe "prompt"`
   - Tokenize prompt, run autoregressive loop for 20 tokens, print decoded text continuation.

6. **Size strategy**
   - Dense C style, short names, static buffers, no fancy error text.
   - No dynamic JSON/protobuf, no external libs.
   - Keep within requested compactness target as far as possible.

---

```c
// /app/gpt2.c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <math.h>

#define V 50257
#define N 768
#define L 12
#define H 12
#define M 3072
#define T 1024
#define HS (N/H)
#define KMAX 50000
#define S(x) ((x)<0?0:(x))
typedef struct{char*s;int id;}KV;
static KV kv[KMAX]; static int nk;
static char *id2s[V];
static char *mg[KMAX]; static int nmg;

static float *wte,*wpe,*lnfw,*lnfb,*lmh;
static float *l1w[L],*l1b[L],*attw[L],*attb[L],*projw[L],*projb[L],*l2w[L],*l2b[L],*fcw[L],*fcb[L],*fpw[L],*fpb[L];

static unsigned int hsh(const char*s){unsigned int h=2166136261u;for(;*s;s++)h=(h^*(unsigned char*)s)*16777619u;return h;}
static void put(char*s,int id){if(nk<KMAX){kv[nk].s=strdup(s);kv[nk].id=id;nk++;}}
static int get(const char*s){for(int i=0;i<nk;i++)if(!strcmp(kv[i].s,s))return kv[i].id;return -1;}

static float* rd(FILE*f,long n){float*p=malloc(4*n);if(fread(p,4,n,f)!=(size_t)n)exit(1);return p;}

static void ln(float*y,float*x,float*w,float*b){
  float m=0,v=0; for(int i=0;i<N;i++)m+=x[i]; m/=N;
  for(int i=0;i<N;i++){float d=x[i]-m;v+=d*d;} v=sqrtf(v/N+1e-5f);
  for(int i=0;i<N;i++)y[i]=((x[i]-m)/v)*w[i]+b[i];
}
static float gelu(float x){return .5f*x*(1.f+tanhf(.79788456f*(x+.044715f*x*x*x)));}

int main(int c,char**v){
  if(c<4)return 1;
  FILE*f=fopen(v[1],"rb"); if(!f)return 1;

  wte=rd(f,(long)V*N); wpe=rd(f,(long)T*N);
  for(int l=0;l<L;l++){
    l1w[l]=rd(f,N); l1b[l]=rd(f,N);
    attw[l]=rd(f,(long)N*3*N); attb[l]=rd(f,3*N);
    projw[l]=rd(f,(long)N*N); projb[l]=rd(f,N);
    l2w[l]=rd(f,N); l2b[l]=rd(f,N);
    fcw[l]=rd(f,(long)N*M); fcb[l]=rd(f,M);
    fpw[l]=rd(f,(long)M*N); fpb[l]=rd(f,N);
  }
  lnfw=rd(f,N); lnfb=rd(f,N); lmh=wte; fclose(f);

  FILE*b=fopen(v[2],"r"); if(!b)return 1;
  char line[4096]; int id=0;
  while(fgets(line,4096,b)&&id<V){line[strcspn(line,"\n")]=0; put(line,id); id2s[id]=strdup(line); id++;}
  while(fgets(line,4096,b)&&nmg<KMAX){line[strcspn(line,"\n")]=0; mg[nmg++]=strdup(line);}
  fclose(b);

  int tok[T],nt=0;
  char *p=v[3],tmp[4096]; strncpy(tmp,p,4095); tmp[4095]=0;
  char *s=strtok(tmp," ");
  while(s&&nt<T){int i=get(s); tok[nt++]=(i>=0?i:0); s=strtok(NULL," ");}

  float x[N],a[N],q[N],k[T][N],vv[T][N],o[N],h1[M],h2[N],res[N],all[T][N];
  for(int step=0;step<20;step++){
    int t=nt-1;
    for(int i=0;i<N;i++) x[i]=wte[tok[t]*N+i]+wpe[t*N+i];
    for(int l=0;l<L;l++){
      ln(a,x,l1w[l],l1b[l]);
      for(int i=0;i<3*N;i++){float z=attb[l][i];for(int j=0;j<N;j++)z+=a[j]*attw[l][j*3*N+i]; if(i<N)q[i]=z; else if(i<2*N)k[t][i-N]=z; else vv[t][i-2*N]=z;}
      for(int i=0;i<N;i++)o[i]=0;
      for(int h=0;h<H;h++){
        float sc[T],mx=-1e30f,sm=0;
        for(int tt=0;tt<=t;tt++){float z=0;for(int j=0;j<HS;j++)z+=q[h*HS+j]*k[tt][h*HS+j];z/=sqrtf(HS);sc[tt]=z;if(z>mx)mx=z;}
        for(int tt=0;tt<=t;tt++){sc[tt]=expf(sc[tt]-mx);sm+=sc[tt];}
        for(int j=0;j<HS;j++){float z=0;for(int tt=0;tt<=t;tt++)z+=sc[tt]/sm*vv[tt][h*HS+j];o[h*HS+j]=z;}
      }
      for(int i=0;i<N;i++){float z=projb[l][i];for(int j=0;j<N;j++)z+=o[j]*projw[l][j*N+i];res[i]=x[i]+z;}
      ln(a,res,l2w[l],l2b[l]);
      for(int i=0;i<M;i++){float z=fcb[l][i];for(int j=0;j<N;j++)z+=a[j]*fcw[l][j*M+i];h1[i]=gelu(z);}
      for(int i=0;i<N;i++){float z=fpb[l][i];for(int j=0;j<M;j++)z+=h1[j]*fpw[l][j*N+i];x[i]=res[i]+z;}
    }
    ln(a,x,lnfw,lnfb);
    int best=0; float bv=-1e30f;
    for(int id2=0;id2<V;id2++){float z=0;for(int j=0;j<N;j++)z+=a[j]*lmh[id2*N+j]; if(z>bv){bv=z;best=id2;}}
    tok[nt++]=best;
    printf("%s",id2s[best]?id2s[best]:"");
  }
  return 0;
}
```
