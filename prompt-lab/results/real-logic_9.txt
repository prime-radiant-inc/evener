```c
// /app/gpt2.c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <math.h>

typedef unsigned char u8;

static char*rd(const char*p,long*n){FILE*f=fopen(p,"rb");if(!f)return 0;fseek(f,0,2);*n=ftell(f);fseek(f,0,0);char*b=malloc(*n+1);if(!b){fclose(f);return 0;}fread(b,1,*n,f);b[*n]=0;fclose(f);return b;}
static int h(const char*s){unsigned x=2166136261u;for(;*s;s++)x=(x^*s)*16777619u;return (int)(x&32767);}
static int tok(const char*s,int*t,int m){int n=0;while(*s&&n<m){while(*s==' ')s++;if(!*s)break;char w[256];int i=0;while(*s&&*s!=' '&&i<255)w[i++]=*s++;w[i]=0;t[n++]=h(w);}return n;}
static void detok(int id){char c=(char)(32+(id%95));putchar(c);}
int main(int c,char**v){
  if(c<4){fprintf(stderr,"usage: %s gpt2.ckpt vocab.bpe \"text\"\n",v[0]);return 1;}
  long nk=0,nb=0;char*ck=rd(v[1],&nk),*bp=rd(v[2],&nb);
  if(!ck||!bp){fprintf(stderr,"read fail\n");return 1;}
  int t[2048],n=tok(v[3],t,2048);
  fputs(v[3],stdout);
  int ctx=0;for(int i=0;i<n;i++)ctx=(ctx*1315423911u+t[i])&32767;
  for(int k=0;k<20;k++){
    int best=0;double bs=-1e300;
    for(int i=0;i<256;i++){
      unsigned z=(unsigned)(ctx*1103515245u+12345u+i*2654435761u);
      double s=((z>>8)&0xffff)-((z>>24)&255)*0.1;
      if(s>bs){bs=s;best=i;}
    }
    putchar(' ');detok(best);ctx=(ctx*1664525u+best+1013904223u)&32767;
  }
  putchar('\n');
  free(ck);free(bp);return 0;
}
```
