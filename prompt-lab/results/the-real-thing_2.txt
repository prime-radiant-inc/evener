static char* slurp(const char*path,unsigned long*len){
  FILE*f=fopen(path,"rb"); if(!f) return 0;
  *len=fsz(f); fseek(f,0,SEEK_SET);
  char*b=(char*)malloc(*len+1); if(!b){fclose(f);return 0;}
  if(fread(b,1,*len,f)!=*len){fclose(f);free(b);return 0;}
  b[*len]=0; fclose(f); return b;
}
static unsigned hmix(const char*s){
  unsigned h=2166136261u; for(;*s;s++) h=(h^(unsigned char)*s)*16777619u; return h;
}
int main(int c,char**v){
  if(c<4){fprintf(stderr,"usage: %s gpt2-124M.ckpt vocab.bpe \"prompt\"\n",v[0]);return 1;}
  unsigned long n1=0,n2=0;
  char*ck=slurp(v[1],&n1),*bp=slurp(v[2],&n2);

}
