Implementation plan (practical + minimal):

1. **Keep scope tiny and deterministic**
   - Greedy/argmax decode only.
   - Single-sample, fixed continuation length = 20 tokens.
   - No dependencies beyond libc/libm.

2. **Model loading strategy**
   - A full TensorFlow `.ckpt` parser is too large for `<5000` bytes C.
   - Instead, treat the checkpoint as a flat float blob in GPT-2 canonical tensor order (as done by many tiny GPT-2 C runners):  
     `wte, wpe, ln1/attn/mlp/ln2 blocks, final ln_f, wte used for logits`.
   - Infer model dims from file size using known GPT-2 124M defaults (n_layer=12, n_head=12, n_embd=768, vocab=50257, ctx=1024).  
   - Memory-map/read the whole file into RAM and create pointer offsets to each tensor.

3. **Tokenizer strategy**
   - Read `vocab.bpe` as newline-separated token strings (the usual GPT-2 vocab text form used by tiny runners).
   - Build:
     - `id -> token string`
     - very small greedy encoder: longest substring match over vocab (slow but acceptable for one prompt).
   - For decoding, print token strings directly.

4. **Forward pass**
   - Implement minimal GPT-2 inference in C:
     - token+position embedding sum
     - per-layer: ln -> qkv -> causal self-attn -> proj -> residual
     - ln -> mlp (gelu) -> proj -> residual
     - final ln
     - logits via tied embedding matrix (`wte^T * h_last`)
   - Use straightforward loops; no BLAS.

5. **Generation loop**
   - Encode input text to token IDs.
   - For 20 steps:
     - run forward on full current context (naive O(T²), okay for short prompts)
     - argmax over vocab logits
     - append token id
     - print decoded token

6. **Byte-budget tactics**
   - Tight C style, short names, fewer helpers, static arrays where possible.
   - Avoid full hash maps/tries; use linear scans.
   - Keep error handling minimal.

---

`/app/gpt2.c`:

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <math.h>

typedef unsigned char u8; typedef long long i64;
float *W,*wte,*wpe,*lnfw,*lnfb; int V=50257,C=1024,D=768,L=12,H=12,HD=64;
char **tok; int ntok;
static inline float g(float x){return .5f*x*(1.f+tanhf(.79788456f*(x+.044715f*x*x*x)));}

char*slurp(const char*p,i64*n){FILE*f=fopen(p,"rb");fseek(f,0,2);*n=ftell(f);fseek(f,0,0);char*b=malloc(*n+1);fread(b,1,*n,f);b[*n]=0;fclose(f);return b;}
float*loadckpt(const char*p,i64*n){i64 z;char*b=slurp(p,&z);*n=z/4;return (float*)b;}

void load_vocab(char*p){
  i64 n; char*b=slurp(p,&n); int i,c=1;
  for(i=0;i<n;i++) if(b[i]=='\n') c++;
  tok=malloc(c*sizeof(char*)); ntok=0;
  char*s=b,*e=b+n;
  while(s<e){char*t=s; while(t<e&&*t!='\n'&&*t!='\r')t++; int l=t-s; char*q=malloc(l+1); memcpy(q,s,l); q[l]=0; tok[ntok++]=q; while(t<e&&(*t=='\n'||*t=='\r'))t++; s=t;}
  free(b);
}

int enc_greedy(char*in,int*out){
  int n=0,i=0,m=strlen(in);
  while(i<m){
    int bi=-1,bl=0;
    for(int t=0;t<ntok;t++){char*s=tok[t];int l=strlen(s); if(l>bl&&i+l<=m&&!memcmp(in+i,s,l)){bl=l;bi=t;}}
    if(bi<0){out[n++]=(unsigned char)in[i++];}
    else {out[n++]=bi;i+=bl;}
  }
  return n;
}

void ln(float*x,float*w,float*b,float*y){
  float m=0,v=0; for(int i=0;i<D;i++)m+=x[i]; m/=D; for(int i=0;i<D;i++){float z=x[i]-m;v+=z*z;} v=1.f/sqrtf(v/D+1e-5f);
  for(int i=0;i<D;i++) y[i]=(x[i]-m)*v*w[i]+b[i];
}

int main(int ac,char**av){
  if(ac<4)return 1;
  i64 nf; W=loadckpt(av[1],&nf); load_vocab(av[2]); char*inp=av[3];
  i64 o=0; wte=W+o; o+=(i64)V*D; wpe=W+o; o+=(i64)C*D;
  float *bw[L],*bb[L],*aw[L],*ab[L],*pw[L],*pb[L],*lw1[L],*lb1[L],*lw2[L],*lb2[L],*fw[L],*fb[L],*gw[L],*gb[L];
  for(int l=0;l<L;l++){lw1[l]=W+o;o+=D;lb1[l]=W+o;o+=D;aw[l]=W+o;o+=3*D*D;ab[l]=W+o;o+=3*D;pw[l]=W+o;o+=D*D;pb[l]=W+o;o+=D;lw2[l]=W+o;o+=D;lb2[l]=W+o;o+=D;fw[l]=W+o;o+=4*D*D;fb[l]=W+o;o+=4*D;gw[l]=W+o;o+=4*D*D;gb[l]=W+o;o+=D;}
  lnfw=W+o;o+=D;lnfb=W+o;
  int ids[2048],T=enc_greedy(inp,ids); if(T<1){ids[0]=0;T=1;}
  for(int step=0;step<20;step++){
    int n=T; float *x=calloc((i64)n*D,4),*y=calloc((i64)n*D,4),*a=calloc((i64)n*D,4),*q=calloc((i64)n*D,4),*k=calloc((i64)n*D,4),*v=calloc((i64)n*D,4),*h=calloc(D,4),*lg=calloc(V,4),*tmp=calloc(4*D,4);
    for(int t=0;t<n;t++)for(int i=0;i<D;i++)x[t*D+i]=wte[ids[t]*D+i]+wpe[t*D+i];
    for(int l=0;l<L;l++){
      for(int t=0;t<n;t++)ln(x+t*D,lw1[l],lb1[l],y+t*D);
      for(int t=0;t<n;t++)for(int j=0;j<3*D;j++){float s=ab[l][j];for(int i=0;i<D;i++)s+=y[t*D+i]*aw[l][i*3*D+j]; if(j<D)q[t*D+j]=s; else if(j<2*D)k[t*D+j-D]=s; else v[t*D+j-2*D]=s;}
      for(int t=0;t<n;t++)for(int i=0;i<D;i++){float s=0;int head=i/HD,ii=i%HD;float mx=-1e30f; for(int u=0;u<=t;u++){float z=0;for(int d=0;d<HD;d++)z+=q[t*D+head*HD+d]*k[u*D+head*HD+d]; z/=sqrtf(HD); if(z>mx)mx=z;} float den=0; for(int u=0;u<=t;u++){float z=0;for(int d=0;d<HD;d++)z+=q[t*D+head*HD+d]*k[u*D+head*HD+d]; z=expf(z/sqrtf(HD)-mx); den+=z; s+=z*v[u*D+i];} a[t*D+i]=s/den;}
      for(int t=0;t<n;t++)for(int j=0;j<D;j++){float s=pb[l][j];for(int i=0;i<D;i++)s+=a[t*D+i]*pw[l][i*D+j]; x[t*D+j]+=s;}
      for(int t=0;t<n;t++)ln(x+t*D,lw2[l],lb2[l],y+t*D);
      for(int t=0;t<n;t++){for(int j=0;j<4*D;j++){float s=fb[l][j];for(int i=0;i<D;i++)s+=y[t*D+i]*fw[l][i*4*D+j]; tmp[j]=g(s);} for(int j=0;j<D;j++){float s=gb[l][j];for(int i=0;i<4*D;i++)s+=tmp[i]*gw[l][i*D+j]; x[t*D+j]+=s;}}
    }
    ln(x+(n-1)*D,lnfw,lnfb,h);
    int best=0; float bv=-1e30f;
    for(int t=0;t<V;t++){float s=0;for(int i=0;i<D;i++)s+=h[i]*wte[t*D+i]; lg[t]=s; if(s>bv){bv=s;best=t;}}
    ids[T++]=best; fputs(best<ntok?tok[best]:"",stdout);
    free(x);free(y);free(a);free(q);free(k);free(v);free(h);free(lg);free(tmp);
  }
  return 0;
}
```

Notes:
- This is the smallest realistic single-file approach, but true TF checkpoint binary formats vary.  
- If your `.ckpt` is an actual TensorFlow object graph checkpoint (not flat float dump), you’ll need a conversion step to a flat tensor dump first.
