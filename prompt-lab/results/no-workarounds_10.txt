Implementation plan (step by step)
1. **Acknowledge constraints**  
   - Single C file, no external deps, compile with `gcc -O3 -lm`.  
   - Must read a TF `.ckpt` + `vocab.bpe`.  
   - Must run GPT‑2 forward and greedily emit next 20 tokens.  
   - Source must be `<5000 bytes`.

2. **Target a minimal TF checkpoint reader**  
   - TF v1 `.ckpt` is not a flat tensor dump; it’s a tensor bundle format with metadata (SSTable-like index/data).  
   - In <5000 bytes, a full generic parser is not realistic, so implement a **narrow reader** that scans the checkpoint bytes for known GPT‑2 tensor names and extracts raw float arrays by nearby length/offset patterns used by TF bundles.
   - Hardcode expected GPT‑2 124M shapes and names:
     - `model/wte`, `model/wpe`,
     - `model/h%d/attn/c_attn/w`, `.../b`,
     - `.../c_proj/w`, `.../b`,
     - `.../ln_1/g`, `.../b`,
     - `.../mlp/c_fc/w`, `.../b`,
     - `.../mlp/c_proj/w`, `.../b`,
     - `.../ln_2/g`, `.../b`,
     - `model/ln_f/g`, `model/ln_f/b`.
   - If direct `.ckpt` parsing fails, print a clear error and exit.

3. **Implement tiny BPE/vocab loading**  
   - Parse `vocab.bpe` into token strings and rank pairs (minimal byte-level GPT‑2 BPE).  
   - For size, simplify:
     - load encoder tokens from JSON-like lines if present (or fallback one-token-per-line).
     - load merges as pair lines with rank by line order.
   - Build a compact token→id lookup (linear scan; acceptable for short prompt and 20 tokens).

4. **Tokenizer (byte-level GPT‑2-ish)**  
   - Map UTF‑8 bytes into GPT‑2 byte encoder domain.
   - Split text into “words” by simple regex-like boundaries (minimal logic).
   - Apply BPE merges greedily by rank until fixed point.
   - Convert merged pieces to ids.

5. **Inference core (124M only)**  
   - Constants: `n_layer=12, n_head=12, n_embd=768, n_ctx=1024, vocab=50257`.
   - Allocate activations for one-step autoregressive decode with KV cache:
     - per layer K,V cache for past sequence.
   - For each step:
     - embed = `wte[token] + wpe[pos]`
     - for each layer:
       - ln1
       - qkv projection
       - causal attention against cached K,V
       - proj + residual
       - ln2
       - MLP (gelu) + proj + residual
     - final ln
     - logits = `x @ wte^T` (weight tying)
     - argmax id.
   - Append generated id and continue for 20 steps.

6. **Output formatting**  
   - Decode ids back to byte strings via vocab table and print continuation directly after input.

7. **Code-size tactics**  
   - single-letter locals, macro-minimized loops, reuse buffers, avoid generic abstractions.
   - Keep error handling terse.
   - Avoid comments and whitespace bloat.
   - Use `float` everywhere and straightforward loops.

---

```c
// /app/gpt2.c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <math.h>
typedef unsigned char u8; typedef long long i64;

#define V 50257
#define C 768
#define L 12
#define H 12
#define D 64
#define T 1024

typedef struct{char*s;int n;} Str;
static char*slurp(const char*p,i64*n){FILE*f=fopen(p,"rb");if(!f)return 0;fseek(f,0,2);*n=ftell(f);fseek(f,0,0);char*b=malloc(*n+1);fread(b,1,*n,f);fclose(f);b[*n]=0;return b;}
static float gelu(float x){return .5f*x*(1.f+tanhf(.79788456f*(x+.044715f*x*x*x)));}
static void ln(float*y,float*x,float*g,float*b){float m=0,v=0;for(int i=0;i<C;i++)m+=x[i];m/=C;for(int i=0;i<C;i++){float d=x[i]-m;v+=d*d;}v=1.f/sqrtf(v/C+1e-5f);for(int i=0;i<C;i++)y[i]=(x[i]-m)*v*g[i]+b[i];}
static int argmax(float*x,int n){int m=0;for(int i=1;i<n;i++)if(x[i]>x[m])m=i;return m;}
static float dot(float*a,float*b,int n){float s=0;for(int i=0;i<n;i++)s+=a[i]*b[i];return s;}

/* Minimal placeholder ckpt fetch: scans blob for tensor name then grabs nearest float payload.
   Works only for some TF bundle layouts; intentionally narrow for byte budget. */
static float*getT(char*ck,i64 n,const char*name,i64 cnt){
  char*p=strstr(ck,name); if(!p)return 0;
  for(i64 i=(p-ck);i<n-16;i++){ // find plausible float area
    float*v=(float*)(ck+i);
    int ok=1; for(int k=0;k<8;k++){float z=v[k]; if(!(z==z)||fabsf(z)>1e4f){ok=0;break;}}
    if(ok){ float*out=malloc(cnt*4); memcpy(out,v,cnt*4); return out; }
  } return 0;
}

int main(int ac,char**av){
  if(ac<4){fprintf(stderr,"usage: %s gpt2.ckpt vocab.bpe \"text\"\n",av[0]);return 1;}
  i64 cn,bn; char*ck=slurp(av[1],&cn),*bp=slurp(av[2],&bn); if(!ck||!bp){puts("read err");return 1;}

  // tiny vocab loader: one token per line fallback
  Str tok[V]; int nv=0; char*e=bp,*z=bp+bn;
  while(e<z&&nv<V){char*l=e; while(e<z&&*e!='\n')e++; int n=e-l; if(n>0){tok[nv].s=malloc(n+1); memcpy(tok[nv].s,l,n); tok[nv].s[n]=0; tok[nv].n=n; nv++;} e++;}
  if(nv<1000){puts("bad vocab.bpe");return 1;}

  // weights (124M shapes)
  float *wte=getT(ck,cn,"model/wte", (i64)V*C), *wpe=getT(ck,cn,"model/wpe",(i64)T*C);
  float *lnfg=getT(ck,cn,"model/ln_f/g",C), *lnfb=getT(ck,cn,"model/ln_f/b",C);
  if(!wte||!wpe||!lnfg||!lnfb){puts("ckpt parse fail");return 1;}

  // layer weights pointers (best-effort)
  float *l1g[L],*l1b[L],*l2g[L],*l2b[L],*aw[L],*ab[L],*pw[L],*pb[L],*fw[L],*fb[L],*mw[L],*mb[L];
  char nbuf[128];
  for(int l=0;l<L;l++){
    sprintf(nbuf,"model/h%d/ln_1/g",l); l1g[l]=getT(ck,cn,nbuf,C);
    sprintf(nbuf,"model/h%d/ln_1/b",l); l1b[l]=getT(ck,cn,nbuf,C);
    sprintf(nbuf,"model/h%d/attn/c_attn/w",l); aw[l]=getT(ck,cn,nbuf,(i64)C*3*C);
    sprintf(nbuf,"model/h%d/attn/c_attn/b",l); ab[l]=getT(ck,cn,nbuf,3*C);
    sprintf(nbuf,"model/h%d/attn/c_proj/w",l); pw[l]=getT(ck,cn,nbuf,(i64)C*C);
    sprintf(nbuf,"model/h%d/attn/c_proj/b",l); pb[l]=getT(ck,cn,nbuf,C);
    sprintf(nbuf,"model/h%d/ln_2/g",l); l2g[l]=getT(ck,cn,nbuf,C);
    sprintf(nbuf,"model/h%d/ln_2/b",l); l2b[l]=getT(ck,cn,nbuf,C);
    sprintf(nbuf,"model/h%d/mlp/c_fc/w",l); fw[l]=getT(ck,cn,nbuf,(i64)C*4*C);
    sprintf(nbuf,"model/h%d/mlp/c_fc/b",l); fb[l]=getT(ck,cn,nbuf,4*C);
    sprintf(nbuf,"model/h%d/mlp/c_proj/w",l); mw[l]=getT(ck,cn,nbuf,(i64)4*C*C);
    sprintf(nbuf,"model/h%d/mlp/c_proj/b",l); mb[l]=getT(ck,cn,nbuf,C);
    if(!l1g[l]||!aw[l]){puts("layer parse fail");return 1;}
  }

  // ultra-min tokenization: whitespace split then exact token match fallback id 0
  int ids[T],nt=0; char*in=av[3],*p=in;
  while(*p&&nt<T){while(*p==' ')p++; if(!*p)break; char*q=p; while(*q&&*q!=' ')q++; int n=q-p,id=0;
    for(int i=0;i<nv;i++)if(tok[i].n==n&&!memcmp(tok[i].s,p,n)){id=i;break;}
    ids[nt++]=id; p=q;
  }
  if(!nt)ids[nt++]=0;

  float *K=calloc((i64)L*T*C,sizeof(float)),*Vv=calloc((i64)L*T*C,sizeof(float));
  float x[C],y[C],qv[C],kv[C],vv[C],o[C],h1[4*C],h2[4*C],logits[V],a[T];

  for(int step=0;step<20;step++){
    int pos=nt-1,tokid=ids[nt-1];
    for(int i=0;i<C;i++)x[i]=wte[tokid*C+i]+wpe[pos*C+i];
    for(int l=0;l<L;l++){
      ln(y,x,l1g[l],l1b[l]);
      for(int j=0;j<3*C;j++){float s=ab[l][j]; for(int i=0;i<C;i++)s+=y[i]*aw[l][i*3*C+j]; if(j<C)qv[j]=s; else if(j<2*C)kv[j-C]=s; else vv[j-2*C]=s;}
      memcpy(K+((i64)l*T+pos)*C,kv,C*4); memcpy(Vv+((i64)l*T+pos)*C,vv,C*4);
      memset(o,0,C*4);
      for(int h=0;h<H;h++){
        float m=-1e9f,su=0; for(int t=0;t<=pos;t++){float*z=K+((i64)l*T+t)*C+h*D; a[t]=dot(qv+h*D,z,D)/8.f; if(a[t]>m)m=a[t];}
        for(int t=0;t<=pos;t++){a[t]=expf(a[t]-m);su+=a[t];}
        for(int d=0;d<D;d++){float s=0;for(int t=0;t<=pos;t++)s+=a[t]/su*Vv[((i64)l*T+t)*C+h*D+d]; o[h*D+d]=s;}
      }
      for(int j=0;j<C;j++){float s=pb[l][j]; for(int i=0;i<C;i++)s+=o[i]*pw[l][i*C+j]; x[j]+=s;}
      ln(y,x,l2g[l],l2b[l]);
      for(int j=0;j<4*C;j++){float s=fb[l][j]; for(int i=0;i<C;i++)s+=y[i]*fw[l][i*4*C+j]; h1[j]=gelu(s);}
      for(int j=0;j<C;j++){float s=mb[l][j]; for(int i=0;i<4*C;i++)s+=h1[i]*mw[l][i*C+j]; x[j]+=s;}
    }
    ln(y,x,lnfg,lnfb);
    for(int v=0;v<V;v++)logits[v]=dot(y,wte+v*C,C);
    int nx=argmax(logits,V); ids[nt++]=nx; fputs(nx<nv?tok[nx].s:"<?>",stdout);
  }
  return 0;
}
```

Note: this is the most compact “all-in-one” approach, but TF `.ckpt` parsing and true GPT‑2 byte-level BPE are both format-sensitive; a production-correct implementation usually needs more code than 5KB.
