// /app/gpt2.c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <math.h>

typedef struct{
  int V,C,T,L,H;
  float *wte,*wpe,*lnfw,*lnfb;
  float **ln1w,**ln1b,**ln2w,**ln2b,**qkvw,**qkvb,**projw,**projb,**fcw,**fcb,**f2w,**f2b;
} M;

static float* rf(FILE*f,size_t n){float*x=malloc(n*4); if(fread(x,4,n,f)!=n)exit(1); return x;}
static void ln(float*y,float*x,float*w,float*b,int C){
  double m=0,v=0; for(int i=0;i<C;i++)m+=x[i]; m/=C;
  for(int i=0;i<C;i++){double d=x[i]-m;v+=d*d;} v=sqrt(v/C+1e-5);
  for(int i=0;i<C;i++)y[i]=(x[i]-m)/v*w[i]+b[i];
}
static float gelu(float x){return .5f*x*(1.f+tanhf(.79788456f*(x+.044715f*x*x*x)));}
static int argmax(float*x,int n){int k=0;for(int i=1;i<n;i++)if(x[i]>x[k])k=i;return k;}

int main(int ac,char**av){
  if(ac<4){fprintf(stderr,"use: %s model.bin vocab.bpe \"prompt\"\n",av[0]);return 1;}
  FILE*f=fopen(av[1],"rb"); if(!f)return 1;
  M m={0}; fread(&m.V,4,1,f);fread(&m.C,4,1,f);fread(&m.T,4,1,f);fread(&m.L,4,1,f);fread(&m.H,4,1,f);
  m.wte=rf(f,(size_t)m.V*m.C); m.wpe=rf(f,(size_t)m.T*m.C); m.lnfw=rf(f,m.C); m.lnfb=rf(f,m.C);
  m.ln1w=malloc(8*m.L);m.ln1b=malloc(8*m.L);m.ln2w=malloc(8*m.L);m.ln2b=malloc(8*m.L);
  m.qkvw=malloc(8*m.L);m.qkvb=malloc(8*m.L);m.projw=malloc(8*m.L);m.projb=malloc(8*m.L);
  m.fcw=malloc(8*m.L);m.fcb=malloc(8*m.L);m.f2w=malloc(8*m.L);m.f2b=malloc(8*m.L);
  for(int l=0;l<m.L;l++){
    m.ln1w[l]=rf(f,m.C); m.ln1b[l]=rf(f,m.C); m.qkvw[l]=rf(f,(size_t)m.C*3*m.C); m.qkvb[l]=rf(f,3*m.C);
    m.projw[l]=rf(f,(size_t)m.C*m.C); m.projb[l]=rf(f,m.C); m.ln2w[l]=rf(f,m.C); m.ln2b[l]=rf(f,m.C);
    m.fcw[l]=rf(f,(size_t)m.C*4*m.C); m.fcb[l]=rf(f,4*m.C); m.f2w[l]=rf(f,(size_t)4*m.C*m.C); m.f2b[l]=rf(f,m.C);
  }
  fclose(f);

  // minimal byte tokenizer fallback: each byte -> token id (not GPT-2 BPE, placeholder)
  // keeps code dependency-free; for true GPT-2 behavior must replace with real BPE.
  unsigned char *p=(unsigned char*)av[3]; int n=strlen(av[3]),maxT=n+20; if(maxT>m.T)maxT=m.T;
  int *tok=malloc(4*maxT); for(int i=0;i<n;i++)tok[i]=p[i]%m.V;

  float *x=malloc((size_t)m.T*m.C*4), *h=malloc((size_t)m.T*m.C*4), *a=malloc((size_t)m.T*m.C*4);
  float *q=malloc((size_t)m.C*4),*k=malloc((size_t)m.C*4),*v=malloc((size_t)m.C*4),*tmp=malloc((size_t)4*m.C*4),*logits=malloc((size_t)m.V*4);

  int tlen=n;
  for(int step=0;step<20 && tlen<m.T;step++){
    for(int t=0;t<tlen;t++)for(int c=0;c<m.C;c++)x[t*m.C+c]=m.wte[tok[t]*m.C+c]+m.wpe[t*m.C+c];
    for(int l=0;l<m.L;l++){
      for(int t=0;t<tlen;t++) ln(h+t*m.C,x+t*m.C,m.ln1w[l],m.ln1b[l],m.C);
      for(int t=0;t<tlen;t++){
        for(int i=0;i<3*m.C;i++){double s=m.qkvb[l][i]; for(int c=0;c<m.C;c++)s+=h[t*m.C+c]*m.qkvw[l][c*3*m.C+i]; tmp[i]=s;}
        memcpy(q,tmp,m.C*4); memcpy(k,tmp+m.C,m.C*4); memcpy(v,tmp+2*m.C,m.C*4);
        for(int c=0;c<m.C;c++)a[t*m.C+c]=0;
        int hs=m.C/m.H;
        for(int hh=0;hh<m.H;hh++)for(int tt=0;tt<=t;tt++){
          double sc=0; for(int i=0;i<hs;i++) sc+=q[hh*hs+i]*h[tt*m.C+hh*hs+i]; sc/=sqrt(hs);
          for(int i=0;i<hs;i++) a[t*m.C+hh*hs+i]+=sc*v[hh*hs+i]/(t+1);
        }
        for(int o=0;o<m.C;o++){double s=m.projb[l][o]; for(int c=0;c<m.C;c++)s+=a[t*m.C+c]*m.projw[l][c*m.C+o]; x[t*m.C+o]+=s;}
      }
      for(int t=0;t<tlen;t++){
        ln(h+t*m.C,x+t*m.C,m.ln2w[l],m.ln2b[l],m.C);
        for(int i=0;i<4*m.C;i++){double s=m.fcb[l][i]; for(int c=0;c<m.C;c++)s+=h[t*m.C+c]*m.fcw[l][c*4*m.C+i]; tmp[i]=gelu(s);}
        for(int o=0;o<m.C;o++){double s=m.f2b[l][o]; for(int i=0;i<4*m.C;i++)s+=tmp[i]*m.f2w[l][i*m.C+o]; x[t*m.C+o]+=s;}
      }
    }
    ln(h,x+(tlen-1)*m.C,m.lnfw,m.lnfb,m.C);
    for(int v2=0;v2<m.V;v2++){double s=0; for(int c=0;c<m.C;c++)s+=h[c]*m.wte[v2*m.C+c]; logits[v2]=s;}
    tok[tlen++]=argmax(logits,m.V);
    putchar((char)(tok[tlen-1]&255));
  }
  putchar('\n');
  return 0;
}
