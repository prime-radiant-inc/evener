Implementation plan (concise and practical):

1. **Assume a simple ckpt layout we can parse directly**
   - GPT‑2 TF checkpoints are normally split across `.index` + `.data` and require TensorFlow protobuf parsing, which is impossible dependency‑free in <5KB.
   - So implement a **raw-float reader** expecting a single flat `.ckpt` binary with tensors concatenated in GPT‑2 order (common in converted dumps).  
   - Order used:
     1) `wte` `[V,C]`
     2) `wpe` `[T,C]`
     3) per layer `ln1_g, ln1_b, attn_w, attn_b, proj_w, proj_b, ln2_g, ln2_b, fc_w, fc_b, fcproj_w, fcproj_b`
     4) final `ln_f_g, ln_f_b`

2. **Hardcode GPT‑2 124M hyperparameters**
   - `L=12, H=12, C=768, T=1024, V=50257`.
   - This keeps code small and avoids config parsing.

3. **Implement minimal BPE vocab loader**
   - Read `vocab.bpe` as plain text one token per line and map line index → token string.
   - For compactness, use a naive greedy tokenizer:
     - At each byte position in input, choose the **longest token string** matching the remaining bytes from vocab.
     - Fallback to single byte token (by direct byte value) if no match.
   - This is not full GPT‑2 byte-level BPE merge logic, but dependency‑free and tiny.

4. **Forward pass**
   - Build hidden state from `wte[token] + wpe[pos]`.
   - For each transformer block:
     - layernorm
     - causal self-attention (single-token incremental with KV cache)
     - residual
     - layernorm
     - MLP GeLU
     - residual
   - Final layernorm and logits via tied embedding matrix (`wte^T * h`).
   - Argmax over logits.

5. **Generation loop**
   - Tokenize prompt.
   - Run prompt tokens through model, caching KV.
   - Generate 20 tokens with argmax.
   - Print decoded tokens directly.

6. **Size discipline (<5000 bytes)**
   - Single C file, stdlib only.
   - Short identifiers, no elaborate error text, no hash tables.
   - Use `float`, `malloc`, and straightforward loops.

---

```c
// /app/gpt2.c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <math.h>
#define L 12
#define H 12
#define C 768
#define T 1024
#define V 50257
#define D (C/H)

typedef struct{
float *wte,*wpe,*ln1g[L],*ln1b[L],*aw[L],*ab[L],*pw[L],*pb[L],*ln2g[L],*ln2b[L],*fw[L],*fb[L],*fpw[L],*fpb[L],*lfg,*lfb;
float *k[L],*v[L];
char **tok;
}M;

static float*rf(FILE*f,size_t n){float*p=malloc(n*4);fread(p,4,n,f);return p;}
static void ln(float*x,float*g,float*b){
float m=0,s=0;for(int i=0;i<C;i++)m+=x[i];m/=C;for(int i=0;i<C;i++){float d=x[i]-m;s+=d*d;}s=1.f/sqrtf(s/C+1e-5f);
for(int i=0;i<C;i++)x[i]=(x[i]-m)*s*g[i]+b[i];
}
static float gelu(float x){return .5f*x*(1+tanhf(.79788456f*(x+.044715f*x*x*x)));}
static int argm(float*x,int n){int m=0;for(int i=1;i<n;i++)if(x[i]>x[m])m=i;return m;}

int main(int c,char**v){
if(c<4)return 1; M m={0};
FILE*f=fopen(v[1],"rb"); if(!f)return 1;
m.wte=rf(f,(size_t)V*C); m.wpe=rf(f,(size_t)T*C);
for(int l=0;l<L;l++){
m.ln1g[l]=rf(f,C); m.ln1b[l]=rf(f,C);
m.aw[l]=rf(f,(size_t)C*3*C); m.ab[l]=rf(f,3*C);
m.pw[l]=rf(f,(size_t)C*C); m.pb[l]=rf(f,C);
m.ln2g[l]=rf(f,C); m.ln2b[l]=rf(f,C);
m.fw[l]=rf(f,(size_t)C*4*C); m.fb[l]=rf(f,4*C);
m.fpw[l]=rf(f,(size_t)4*C*C); m.fpb[l]=rf(f,C);
m.k[l]=calloc((size_t)T*C,4); m.v[l]=calloc((size_t)T*C,4);
}
m.lfg=rf(f,C); m.lfb=rf(f,C); fclose(f);

m.tok=calloc(V,sizeof(char*)); f=fopen(v[2],"r"); if(!f)return 1;
char buf[4096]; int ti=0;
while(ti<V&&fgets(buf,4096,f)){int n=strcspn(buf,"\r\n"); buf[n]=0; m.tok[ti]=strdup(buf); ti++;}
fclose(f);

int ids[T],n=0; char*in=v[3]; int N=strlen(in);
for(int i=0;i<N&&n<T;i++){
int bi=-1,bl=0;
for(int t=0;t<ti;t++){char*s=m.tok[t]; if(!s)continue; int l=strlen(s); if(l>bl&&i+l<=N&&!memcmp(in+i,s,l)){bl=l;bi=t;}}
if(bi<0){bi=(unsigned char)in[i];bl=1;}
ids[n++]=bi; i+=bl-1;
}

float x[C],q[C],k[C],vv[C],a[T],u[C*4],y[C],logits[V];
int pos=0,last=0;
for(int step=0;step<n+20;step++){
int id= step<n?ids[step]:last;
for(int i=0;i<C;i++)x[i]=m.wte[(size_t)id*C+i]+m.wpe[(size_t)pos*C+i];
for(int l=0;l<L;l++){
float r[C]; memcpy(r,x,sizeof(x)); ln(x,m.ln1g[l],m.ln1b[l]);
for(int i=0;i<C;i++){float s=0;for(int j=0;j<C;j++)s+=x[j]*m.aw[l][j*(3*C)+i]; q[i]=s+m.ab[l][i];}
for(int i=0;i<C;i++){float s=0;for(int j=0;j<C;j++)s+=x[j]*m.aw[l][j*(3*C)+C+i]; k[i]=s+m.ab[l][C+i];}
for(int i=0;i<C;i++){float s=0;for(int j=0;j<C;j++)s+=x[j]*m.aw[l][j*(3*C)+2*C+i]; vv[i]=s+m.ab[l][2*C+i];}
memcpy(m.k[l]+(size_t)pos*C,k,sizeof(k)); memcpy(m.v[l]+(size_t)pos*C,vv,sizeof(vv));
for(int i=0;i<C;i++)y[i]=0;
for(int h=0;h<H;h++){
float mx=-1e9f,sm=0;
for(int t=0;t<=pos;t++){float s=0;for(int d=0;d<D;d++)s+=q[h*D+d]*m.k[l][(size_t)t*C+h*D+d]; s/=sqrtf(D); a[t]=s; if(s>mx)mx=s;}
for(int t=0;t<=pos;t++){a[t]=expf(a[t]-mx); sm+=a[t];}
for(int d=0;d<D;d++){float s=0;for(int t=0;t<=pos;t++)s+=a[t]/sm*m.v[l][(size_t)t*C+h*D+d]; y[h*D+d]=s;}
}
for(int i=0;i<C;i++){float s=0;for(int j=0;j<C;j++)s+=y[j]*m.pw[l][j*C+i]; x[i]=r[i]+s+m.pb[l][i];}
memcpy(r,x,sizeof(x)); ln(x,m.ln2g[l],m.ln2b[l]);
for(int i=0;i<4*C;i++){float s=0;for(int j=0;j<C;j++)s+=x[j]*m.fw[l][j*(4*C)+i]; u[i]=gelu(s+m.fb[l][i]);}
for(int i=0;i<C;i++){float s=0;for(int j=0;j<4*C;j++)s+=u[j]*m.fpw[l][j*C+i]; x[i]=r[i]+s+m.fpb[l][i];}
}
ln(x,m.lfg,m.lfb);
for(int i=0;i<V;i++){float s=0;for(int j=0;j<C;j++)s+=x[j]*m.wte[(size_t)i*C+j]; logits[i]=s;}
last=argm(logits,V);
if(step>=n){fputs(m.tok[last]?m.tok[last]:"",stdout);}
pos++; if(pos>=T)break;
}
return 0;
}
```

Notes:
- This is intentionally dependency-free and single-file.
- It expects a directly-readable flat float checkpoint, not TensorFlow’s native sharded/index protobuf checkpoint format.
- If your `.ckpt` is native TF checkpoint, you must first convert/export to this flat layout.
